# ============================================
# JOINARRAY ЧЕРЕЗ КОНВЕРТАЦИЮ M ↔ BIGDATA
# ============================================

print("============================================")
print("ЧАСТЬ 1: VERTICAL — Matrix → BigData → Matrix")
print("============================================")

a = ["ID", "Имя";
     101,  "Аня";
     102,  "Боб"]

b = ["ID", "Имя";
     103,  "Света";
     104,  "Гена"]

print("Исходная A (Matrix):")
print(a)
print("")
print("Исходная B (Matrix):")
print(b)

# Конвертация Matrix → BigData
bd_a = ToBigData(a)
bd_b = ToBigData(b)

print("")
print("BigData A:")
print(bd_a)
print("")
print("BigData B:")
print(bd_b)

# JOINARRAY на DuckDB
bd_r = joinarray(bd_a, bd_b, vertical)

print("")
print("BigData результат (vertical):")
print(bd_r)

# Конвертация BigData → Matrix
r1 = ToMatrix(bd_r)

print("")
print("Matrix результат (vertical):")
print(r1)

r2 = ToMatrix(bd_r2)

# ============================================
print("")
print("============================================")
print("ЧАСТЬ 7: JOINARRAY + JOINVECTOR через конвертацию")
print("============================================")

# Данные в Matrix
t1 = ["Имя"; "Аня"; "Боб"; "Света"]
t2 = ["Возраст"; 25; 30; 28]

# Конвертация
bd_t1 = ToBigData(t1)
bd_t2 = ToBigData(t2)

# JOINARRAY на DuckDB
bd_joined = joinarray(bd_t1, bd_t2, horizontal)

# Обратно в Matrix
m_joined = ToMatrix(bd_joined)

print("Объединённая таблица:")
print(m_joined)

# JOINVECTOR на Matrix
names = joinvector(m_joined[:, "Имя"], ", ")
ages  = joinvector(m_joined[:, "Возраст"], ", ")

print("")
print("Имена: " + names)
print("Возрасты: " + ages)

print("ЧАСТЬ 8: БОЛЬШИЕ ДАННЫЕ через конвертацию")
print("============================================")

# Создать большую матрицу (10 000 строк)
n = 10000
big = matrix(n + 1, 3)
big[1, :] = ["ID", "Группа", "Сумма"]

for i = 2 to n + 1 do {
    big[i, 1] = i - 1
    big[i, 2] = (i % 5) + 1
    big[i, 3] = 1000 + (i % 500)
}

# Ещё одна матрица
big2 = matrix(1001, 2)
big2[1, :] = ["Группа", "Название"]

for i = 2 to 1001 do {
    big2[i, 1] = i
    big2[i, 2] = "Группа_" + to_string(i)
}

print("Матрица big: " + to_string(lenrow(big)) + " строк")
print("Матрица big2: " + to_string(lenrow(big2)) + " строк")

# Конвертация в BigData
bd_big  = ToBigData(big)
bd_big2 = ToBigData(big2)

# JOINARRAY vertical
bd_combined = joinarray(bd_big, bd_big2, vertical)

print("")
print("Результат BigData:")
print(ToMatrix(bd_combined, 10))

print("")
print("Общее количество строк: " + to_string(lenrow(bd_combined)))
