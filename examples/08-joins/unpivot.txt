# ============================================================
# UNPIVOT — ТЕСТ БЕЗ ОШИБОК (OK)
# ============================================================
# Все рабочие сценарии — падать НЕ ДОЛЖНО:
#   1.  Базовый: диапазон 2:end + один by
#   2.  names — свои имена колонок
#   3.  Один столбец
#   4.  Диапазон столбцов 2:3
#   5.  end — последний столбец
#   6.  last N — последние N столбцов
#   7.  Несколько by через запятую
#   8.  Несколько by через диапазон
#   9.  Несколько by + names
#   10. Мутация
#   11. Пустая 2D-матрица (только заголовок)
#   12. Один столбец + names
#   13. Два by (Страна, Регион)
#   14. Числовые by-столбцы (по номеру)
#   15. last 1 — один последний столбец
#   16. names с другим регистром / экранирование
#   17. by через номера столбцов
# ============================================================

m = ["Страна", "2021", "2022", "2023";
     "РФ",     146,    144,    143;
     "США",    331,    333,    335;
     "Китай",  1412,   1413,   1411;
     "Индия",  1400,   1405,   1410]

print("=== ИСХОДНАЯ МАТРИЦА ===")
print(m)
print("")


# ------------------------------------------------------------
# 1. Базовый — диапазон 2:end, один by
# ------------------------------------------------------------
print("--- 1. unpivot(m[:, 2:end], by m[:, 'Страна']) ---")
r = unpivot(m[:, 2:end], by m[:, "Страна"])
print(r)
print("")


# ------------------------------------------------------------
# 2. names — свои имена колонок
# ------------------------------------------------------------
print("--- 2. unpivot(..., names 'Год', 'Население') ---")
r = unpivot(m[:, 2:end], by m[:, "Страна"],
            names "Год", "Население")
print(r)
print("")


# ------------------------------------------------------------
# 3. Один столбец
# ------------------------------------------------------------
print("--- 3. unpivot(m[:, 2], by m[:, 'Страна']) ---")
r = unpivot(m[:, 2], by m[:, "Страна"])
print(r)
print("")


# ------------------------------------------------------------
# 4. Диапазон столбцов 2:3
# ------------------------------------------------------------
print("--- 4. unpivot(m[:, 2:3], by m[:, 'Страна']) ---")
r = unpivot(m[:, 2:3], by m[:, "Страна"])
print(r)
print("")


# ------------------------------------------------------------
# 5. end — последний столбец
# ------------------------------------------------------------
print("--- 5. unpivot(m[:, end], by m[:, 'Страна']) ---")
r = unpivot(m[:, end], by m[:, "Страна"])
print(r)
print("")


# ------------------------------------------------------------
# 6. last N — последние N столбцов
# ------------------------------------------------------------
print("--- 6. unpivot(m[:, last 2], by m[:, 'Страна']) ---")
r = unpivot(m[:, last 2], by m[:, "Страна"])
print(r)
print("")


# ------------------------------------------------------------
# 7. Несколько by через запятую
# ------------------------------------------------------------
print("--- 7. unpivot(..., by m[:, 'Страна'], m[:, 'Город']) ---")
m2 = ["Страна", "Город",  "2021", "2022";
      "РФ",     "Москва", 12,     13;
      "РФ",     "СПб",    5,      6;
      "США",    "NY",     8,      9]

r = unpivot(m2[:, 3:end], by m2[:, "Страна"], m2[:, "Город"])
print(r)
print("")


# ------------------------------------------------------------
# 8. Несколько by через диапазон
# ------------------------------------------------------------
print("--- 8. unpivot(..., by m2[:, 1:2]) — диапазон ---")
r = unpivot(m2[:, 3:end], by m2[:, 1:2])
print(r)
print("")


# ------------------------------------------------------------
# 9. Несколько by + names
# ------------------------------------------------------------
print("--- 9. unpivot(..., by ..., ..., names 'Год', 'Значение') ---")
r = unpivot(m2[:, 3:end], by m2[:, "Страна"], m2[:, "Город"],
            names "Год", "Значение")
print(r)
print("")


# ------------------------------------------------------------
# 10. Мутация
# ------------------------------------------------------------
print("--- 10. мутация m3 = unpivot(...) ---")
m3 = ["Страна", "2021", "2022";
      "РФ",     146,    144;
      "США",    331,    333]

m3 = unpivot(m3[:, 2:end], by m3[:, "Страна"])
print(m3)
print("")


# ------------------------------------------------------------
# 11. Пустая 2D-матрица (только заголовок)
# ------------------------------------------------------------
print("--- 11. unpivot пустой 2D-матрицы (только заголовок) ---")
m4 = ["Страна", "2021", "2022";]     # ← 2D, только заголовок

r = unpivot(m4[:, 2:end], by m4[:, "Страна"])
print(r)
print("")
# ОЖИДАНИЕ: только заголовок
#   Страна  Переменная  Значение


# ------------------------------------------------------------
# 12. Один столбец + names
# ------------------------------------------------------------
print("--- 12. unpivot(m[:, 2], ..., names 'Год', 'Население') ---")
r = unpivot(m[:, 2], by m[:, "Страна"],
            names "Год", "Население")
print(r)
print("")


# ------------------------------------------------------------
# 13. Два by — Страна + Регион
# ------------------------------------------------------------
print("--- 13. unpivot(..., by m5[:, 'Страна'], m5[:, 'Регион']) ---")
m5 = ["Страна", "Регион",  "2021", "2022";
      "РФ",     "Европа",  146,    144;
      "США",    "Америка", 331,    333]

r = unpivot(m5[:, 3:end], by m5[:, "Страна"], m5[:, "Регион"])
print(r)
print("")


# ------------------------------------------------------------
# 14. Числовые by-столбцы (по номеру)
# ------------------------------------------------------------
print("--- 14. unpivot(m2[:, 3:end], by m2[:, 1], m2[:, 2]) ---")
r = unpivot(m2[:, 3:end], by m2[:, 1], m2[:, 2])
print(r)
print("")


# ------------------------------------------------------------
# 15. last 1 — один последний столбец
# ------------------------------------------------------------
print("--- 15. unpivot(m[:, last 1], by m[:, 'Страна']) ---")
r = unpivot(m[:, last 1], by m[:, "Страна"])
print(r)
print("")


# ------------------------------------------------------------
# 16. names — свои заголовки для одного столбца
# ------------------------------------------------------------
print("--- 16. unpivot(m[:, 2:3], ..., names 'Year', 'Value') ---")
r = unpivot(m[:, 2:3], by m[:, "Страна"],
            names "Year", "Value")
print(r)
print("")


# ------------------------------------------------------------
# 17. by через номера столбцов (1 и 2)
# ------------------------------------------------------------
print("--- 17. unpivot(m5[:, 3:end], by m5[:, 1], m5[:, 2]) ---")
r = unpivot(m5[:, 3:end], by m5[:, 1], m5[:, 2])
print(r)
print("")


print("=== ВСЕ ТЕСТЫ UNPIVOT (OK) ЗАВЕРШЕНЫ ===")
