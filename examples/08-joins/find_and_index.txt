# ============================================================
# ТЕСТ: find / index
# ============================================================

m = ["Имя", "Отдел", "Возраст";
     "Аня", "IT",    25;
     "Боб", "HR",    32;
     "Света", "IT",  28;
     "Гена", "Sales", 41]
print(m)
# ============================================================
# 1. find — координаты [[строка, столбец], ...]
# ============================================================
print("--- 1. find: координаты ---")
r1 = find(m[:, "Отдел"] == "IT")
print(r1)

# ============================================================
# 2. find(..., rows) — только номера строк
# ============================================================
print("--- 2. find(..., rows) ---")
r2 = find(m[:, "Отдел"] == "IT", rows)
print(r2)

# ============================================================
# 3. find(..., cols) — только номера столбцов
# ============================================================
print("--- 3. find(..., cols) ---")
r3 = find(m[:, "Отдел"] == "IT", cols)
print(r3)

# ============================================================
# 4. find — числовое сравнение
# ============================================================
print("--- 4. find: Возраст > 30 ---")
r4 = find(m[:, "Возраст"] > 30)
print(r4)

# ============================================================
# 5. find — не найдено (пустой результат)
# ============================================================
print("--- 5. find: Sales в Отдел ---")
r5 = find(m[:, "Отдел"] == "Sales")
print(r5)

# ============================================================
# 6. index — алиас find (должен работать так же)
# ============================================================
print("--- 6. index: координаты ---")
r6 = index(m[:, "Отдел"] == "IT")
print(r6)

# ============================================================
# 7. index(..., rows) — алиас с модификатором
# ============================================================
print("--- 7. index(..., rows) ---")
r7 = index(m[:, "Отдел"] == "IT", rows)
print(r7)

# ============================================================
# 8. find — inside (поиск подстроки)
# ============================================================
print("--- 8. find(..., inside) ---")
m2 = ["Имя";
      "Иванов";
      "Петров";
      "Аня";
      "Сидоров"]

r8 = find(m2[:, "Имя"] == "ов", inside)
print(r8)

# ============================================================
# 9. find — ignore (без учёта регистра)
# ============================================================
print("--- 9. find(..., ignore) ---")
m3 = ["Имя";
      "Аня";
      "АНЯ";
      "аня";
      "Боб"]

r9 = find(m3[:, "Имя"] == "аня", ignore)
print(r9)

# ============================================================
# 10. find(..., inside, ignore) — подстрока без регистра
# ============================================================
print("--- 10. find(..., inside, ignore) ---")
m4 = ["Имя";
      "Иванов";
      "ИВАНОВ";
      "Петров";
      "Аня"]

r10 = find(m4[:, "Имя"] == "иван", inside, ignore)
print(r10)

# ============================================================
# 11. find — по вектору
# ============================================================
print("--- 11. find: вектор ---")
v = [10, 20, 30, 40, 50]
r11 = find(v == 30)
print(r11)

# ============================================================
# 12. find — составное условие (and)
# ============================================================
print("--- 12. find: and ---")
r12 = find(m[:, "Отдел"] == "IT" and m[:, "Возраст"] > 26)
print(r12)

# ============================================================
# 13. find — составное условие (or)
# ============================================================
print("--- 13. find: or ---")
r13 = find(m[:, "Отдел"] == "IT" or m[:, "Отдел"] == "HR")
print(r13)

print("")
print("=== ТЕСТ ЗАВЕРШЁН ===")
