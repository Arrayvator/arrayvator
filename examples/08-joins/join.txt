# ============================================
# JOIN НА BIGDATA (DUCKDB) — ВСЕ РЕЖИМЫ
# ============================================

print("============================================")
print("ПОДГОТОВКА: Matrix → BigData")
print("============================================")

# Исходные Matrix
m1 = ["ID", "Имя",   "Отдел_ID";
      101,  "Аня",   1;
      102,  "Боб",   2;
      103,  "Света", 1;
      104,  "Гена",  3]

m2 = ["Отдел_ID", "Название";
      1,          "IT";
      2,          "HR";
      4,          "Sales"]

print("Matrix 1 (сотрудники):")
print(m1)

print("")
print("Matrix 2 (отделы):")
print(m2)

# Конвертация в DuckDB
bd1 = ToBigData(m1)
bd2 = ToBigData(m2)

print("")
print("BigData 1 (сотрудники):")
print(bd1)

print("")
print("BigData 2 (отделы):")
print(bd2)


# ============================================
print("")
print("============================================")
print("1. DUCKDB — LEFT JOIN")
print("============================================")
bd_left = join(bd1, bd2, on "Отдел_ID", how "left")
r1 = ToMatrix(bd_left)
print(r1)


# ============================================
print("")
print("============================================")
print("2. DUCKDB — INNER JOIN")
print("============================================")
bd_inner = join(bd1, bd2, on "Отдел_ID", how "inner")
r2 = ToMatrix(bd_inner)
print(r2)


# ============================================
print("")
print("============================================")
print("3. DUCKDB — RIGHT JOIN")
print("============================================")
bd_right = join(bd1, bd2, on "Отдел_ID", how "right")
r3 = ToMatrix(bd_right)
print(r3)


# ============================================
print("")
print("============================================")
print("4. DUCKDB — OUTER JOIN")
print("============================================")
bd_outer = join(bd1, bd2, on "Отдел_ID", how "outer")
r4 = ToMatrix(bd_outer)
print(r4)


# ============================================
print("")
print("============================================")
print("5. DUCKDB — РАЗНЫЕ ИМЕНА КЛЮЧЕЙ")
print("============================================")
m3 = ["ID", "Имя",   "Код_Отдела";
      101,  "Аня",   1;
      102,  "Боб",   2;
      103,  "Света", 1]

bd3 = ToBigData(m3)
bd_mixed = join(bd3, bd2, on "Код_Отдела" == "Отдел_ID")
r5 = ToMatrix(bd_mixed)
print(r5)


# ============================================
print("")
print("============================================")
print("6. DUCKDB — НЕСКОЛЬКО КЛЮЧЕЙ")
print("============================================")
m4 = ["Год", "Месяц", "Сумма";
      2025,  1,       1000;
      2025,  2,       1500;
      2024,  12,      800]

m5 = ["Год", "Месяц", "План";
      2025,  1,       900;
      2025,  2,       1600;
      2024,  12,      1000]

bd4 = ToBigData(m4)
bd5 = ToBigData(m5)

bd_multi = join(bd4, bd5, on ["Год", "Месяц"])
r6 = ToMatrix(bd_multi)
print(r6)


# ============================================
print("")
print("============================================")
print("7. DUCKDB — КОНФЛИКТ ИМЁН (автосуффиксы)")
print("============================================")
m6 = ["ID", "Имя",  "Дата";
      101,  "Аня",  "2025-01-01";
      102,  "Боб",  "2025-02-01"]

m7 = ["ID", "Дата";
      101,  "2025-03-15";
      102,  "2025-04-20"]

bd6 = ToBigData(m6)
bd7 = ToBigData(m7)

bd_conf = join(bd6, bd7, on "ID")
r7 = ToMatrix(bd_conf)
print(r7)


# ============================================
print("")
print("============================================")
print("8. DUCKDB — СВОИ СУФФИКСЫ")
print("============================================")
bd_suf = join(bd6, bd7, on "ID", suffixes ["_найм", "_увольнение"])
r8 = ToMatrix(bd_suf)
print(r8)


# ============================================
print("")
print("============================================")
print("9. СРАВНЕНИЕ: MATRIX vs DUCKDB (OUTER)")
print("============================================")
r_matrix = join(m1, m2, on "Отдел_ID", how "outer")
r_duckdb = ToMatrix(bd_outer)

print("Matrix OUTER:")
print(r_matrix)

print("")
print("DuckDB OUTER:")
print(r_duckdb)


# ============================================
print("")
print("============================================")
print("10. ПРОВЕРКА: КЛЮЧЕВОЙ СТОЛБЕЦ У RIGHT JOIN")
print("============================================")

print("")
