     "Аня",    "IT",    25,        "Москва",  85000;
     "Боб",    "HR",    32,        "Питер",   95000;
     "Света",  "IT",    28,        "Москва",  70000;
     "Гена",   "Sales", 41,        "Москва",  88000;
     "Даша",   "IT",    35,        "Питер",   62000;
     "Егор",   "HR",    29,        "Москва",  65000;
     "Жанна",  "Sales", 38,        "Питер",   72000;
     "Зина",   "HR",    45,        "Москва",  58000;
     "Игорь",  "IT",    31,        "Питер",   81000;
     "Катя",   "Sales", 27,        "Москва",  80000;
     "Лена",   "IT",    42,        "Питер",   88000;
     "Миша",   "HR",    36,        "Москва",  60000;
     "Нина",   "Sales", 33,        "Питер",   95000;
     "Олег",   "IT",    24,        "Москва",  55000;
     "Петя",   "Sales", 39,        "Питер",   73000]

print(r)
print("")

# 2. Числовое условие
print("=== 2. find(m[:, 'Возраст'] > 35) ===")
r = find(m[:, "Возраст"] > 35)
print(r)
print("")

# 3. По подстроке
print("=== 3. find(m[:, 'Имя'] == 'а', inside) ===")
r = find(m[:, "Имя"] == "а", inside)
print(r)
print("")

# 4. Без регистра
print("=== 4. find(m[:, 'Город'] == 'москва', ignore) ===")
r = find(m[:, "Город"] == "москва", ignore)
print(r)
print("")


# ============================================================
# СЛОЖНЫЕ УСЛОВИЯ: AND
# ============================================================

print("=" * 60)
print("СЛОЖНЫЕ УСЛОВИЯ: AND")
print("=" * 60)
print("")

# 5. AND: IT и возраст > 30
print("=== 5. find(Отдел == 'IT' and Возраст > 30) ===")
r = find(m[:, "Отдел"] == "IT" and m[:, "Возраст"] > 30)
print(r)
# Ожидание: Даша (35), Игорь (31), Лена (42)
print("")

# 6. AND: IT и Москва
print("=== 6. find(Отдел == 'IT' and Город == 'Москва') ===")
r = find(m[:, "Отдел"] == "IT" and m[:, "Город"] == "Москва")
print(r)
# Ожидание: Аня (25), Света (28), Олег (24)
print("")

# 7. AND: Зарплата > 80000 и Возраст < 35
print("=== 7. find(Зарплата > 80000 and Возраст < 35) ===")
r = find(m[:, "Зарплата"] > 80000 and m[:, "Возраст"] < 35)
print(r)
# Ожидание: Аня (25, 85000), Боб (32, 95000), Катя (27, 80000? нет — 80000 не > 80000)
print("")


# ============================================================
# СЛОЖНЫЕ УСЛОВИЯ: OR
# ============================================================

print("=" * 60)
print("СЛОЖНЫЕ УСЛОВИЯ: OR")
print("=" * 60)
print("")

# 8. OR: IT или HR
print("=== 8. find(Отдел == 'IT' or Отдел == 'HR') ===")
r = find(m[:, "Отдел"] == "IT" or m[:, "Отдел"] == "HR")
print(r)
# Ожидание: все IT + все HR
print("")

# 9. OR: Возраст < 26 или Возраст > 40
print("=== 9. find(Возраст < 26 or Возраст > 40) ===")
r = find(m[:, "Возраст"] < 26 or m[:, "Возраст"] > 40)
print(r)
# Ожидание: Аня (25), Гена (41), Зина (45), Лена (42), Олег (24)
print("")


# ============================================================
# СЛОЖНЫЕ УСЛОВИЯ: NOT
# ============================================================

print("=" * 60)
print("СЛОЖНЫЕ УСЛОВИЯ: NOT")
print("=" * 60)
print("")

# 10. NOT: не IT
print("=== 10. find(not (Отдел == 'IT')) ===")
r = find(not (m[:, "Отдел"] == "IT"))
print(r)
# Ожидание: все, кроме IT (HR + Sales)
print("")


# ============================================================
# КОМБИНАЦИИ: AND + OR + NOT
# ============================================================

print("=" * 60)
print("КОМБИНАЦИИ: AND + OR + NOT")
print("=" * 60)
print("")

# 11. AND + OR: (IT или HR) и Возраст > 30
print("=== 11. find((Отдел == 'IT' or Отдел == 'HR') and Возраст > 30) ===")
r = find((m[:, "Отдел"] == "IT" or m[:, "Отдел"] == "HR") and m[:, "Возраст"] > 30)
print(r)
# Ожидание: Боб (32), Даша (35), Зина (45), Игорь (31), Лена (42), Миша (36)
print("")

# 12. AND + NOT: не IT и Возраст > 35
print("=== 12. find(not (Отдел == 'IT') and Возраст > 35) ===")
r = find(not (m[:, "Отдел"] == "IT") and m[:, "Возраст"] > 35)
print(r)
# Ожидание: Гена (41), Жанна (38), Зина (45), Петя (39)
print("")

# 13. Сложное: (IT или Sales) и Город == 'Питер' и Возраст < 40
print("=== 13. find((IT или Sales) и Питер и Возраст < 40) ===")
r = find((m[:, "Отдел"] == "IT" or m[:, "Отдел"] == "Sales") and m[:, "Город"] == "Питер" and m[:, "Возраст"] < 40)
print(r)
# Ожидание: Даша (35, IT, Питер), Жанна (38, Sales, Питер), Игорь (31, IT, Питер), Нина (33, Sales, Питер), Петя (39, Sales, Питер)
print("")
