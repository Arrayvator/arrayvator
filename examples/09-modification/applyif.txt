# ============================================================
# ПРОВЕРКА APPLYIF
# ============================================================
# ЦЕЛЬ: понять, как работает сейчас.
# ============================================================

m = ["Имя", "Отдел", "Статус", "Зарплата";
     "Аня", "IT",    "",       85000;
     "Боб", "HR",    "",       65000;
     "Света","IT",   "",       70000;
     "Гена","Sales", "",       80000;
     "Даша","HR",    "",       72000;
     "Егор","Sales", "",       88000]

print("=== ИСХОДНАЯ МАТРИЦА ===")
print(m)
print("")


# ------------------------------------------------------------
# ТЕСТ 1: базовый — заменить Статус на "VIP" для IT
# ------------------------------------------------------------
print("--- ТЕСТ 1: applyif по отделу IT → VIP ---")
r = applyif(m[:, "Отдел"] == "IT", m[:, "Статус"] = "VIP")
print(r)
print("")
# ОЖИДАНИЕ:
# Аня → VIP (IT)
# Боб → (пусто) (HR)
# Света → VIP (IT)
# Гена → (пусто)
# Даша → (пусто)
# Егор → (пусто)


# ------------------------------------------------------------
# ТЕСТ 2: числовое условие
# ------------------------------------------------------------
print("--- ТЕСТ 2: applyif где ЗП > 80000 → увеличить в 2 раза ---")
r = applyif(m[:, "Зарплата"] > 80000, m[:, "Зарплата"] = 999999)
print(r)
print("")
# ОЖИДАНИЕ:
# Аня (85000) → 999999
# Боб (65000) → 65000
# Света (70000) → 70000
# Гена (80000) → 80000 (не >)
# Даша (72000) → 72000
# Егор (88000) → 999999


# ------------------------------------------------------------
# ТЕСТ 3: составное условие (and)
# ------------------------------------------------------------
print("--- ТЕСТ 3: applyif IT AND возраст > 25 ---")
m2 = ["Имя", "Отдел", "Возраст", "Статус";
      "Аня", "IT",    25,        "";
      "Боб", "IT",    32,        "";
      "Света","IT",   28,        ""]

r = applyif(m2[:, "Отдел"] == "IT" and m2[:, "Возраст"] > 25,
            m2[:, "Статус"] = "VIP")
print(r)
print("")
# ОЖИДАНИЕ:
# Аня (25) → "" (не > 25)
# Боб (32) → VIP
# Света (28) → VIP


# ------------------------------------------------------------
# ТЕСТ 4: присваивание в новый столбец
# ------------------------------------------------------------
print("--- ТЕСТ 4: applyif в новый столбец ---")
r = applyif(m[:, "Отдел"] == "IT", m[:, end+1] = "Айти")
print(r)
print("")
# ОЖИДАНИЕ:
# Новый столбец справа с "Айти" для IT, None для остальных


# ------------------------------------------------------------
# ТЕСТ 5: замена одного значения на другое
# ------------------------------------------------------------
print("--- ТЕСТ 5: applyif 'Sales' → 'Продажи' ---")
r = applyif(m[:, "Отдел"] == "Sales", m[:, "Отдел"] = "Продажи")
print(r)
print("")
# ОЖИДАНИЕ:
# Отдел: Аня=IT, Боб=HR, Света=IT, Гена=Продажи, Даша=HR, Егор=Продажи


# ------------------------------------------------------------
# ТЕСТ 6: мутация исходной матрицы
# ------------------------------------------------------------
print("--- ТЕСТ 6: мутация m ---")
m3 = ["Имя", "Отдел", "Статус";
      "Аня", "IT",    "";
      "Боб", "HR",    ""]

m3 = applyif(m3[:, "Отдел"] == "IT", m3[:, "Статус"] = "VIP")
print(m3)
print("")
# ОЖИДАНИЕ:
# Имя   Отдел  Статус
# Аня   IT     VIP
# Боб   HR     (пусто)
