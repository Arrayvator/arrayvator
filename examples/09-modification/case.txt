

m = ["Имя", "Возраст", "Отдел", "Зарплата";
     "Аня",  25,       "IT",    85000;
     "Боб",  15,       "HR",    65000;
     "Света",32,       "IT",    70000;
     "Гена", 41,       "Sales", 80000;
     "Даша", 17,       "HR",    72000;
     "Егор", 29,       "Sales", 88000]

print("=== ИСХОДНАЯ МАТРИЦА ===")
print(m)
print("")


# ------------------------------------------------------------
# ТЕСТ 1: базовый — по возрасту
# ------------------------------------------------------------
print("--- ТЕСТ 1: case по возрасту ---")
r = case(m[:, "Возраст"],
         when < 18 then "Дитя",
         when < 30 then "Молодой",
         else "Взрослый")
print(r)
print("")
# ОЖИДАНИЕ (вектор):
# ["Молодой", "Дитя", "Взрослый", "Взрослый", "Дитя", "Молодой"]


# ------------------------------------------------------------
# ТЕСТ 2: сравнение со строкой
# ------------------------------------------------------------
print("--- ТЕСТ 2: case по отделу ---")
r = case(m[:, "Отдел"],
         when == "IT" then "Айти",
         else "Другие")
print(r)
print("")
# ОЖИДАНИЕ:
# ["Айти", "Другие", "Айти", "Другие", "Другие", "Другие"]


# ------------------------------------------------------------
# ТЕСТ 3: без else (None по умолчанию)
# ------------------------------------------------------------
print("--- ТЕСТ 3: без else ---")
r = case(m[:, "Возраст"],
         when < 18 then "Ребёнок")
print(r)
print("")
# ОЖИДАНИЕ:
# [None, "Ребёнок", None, None, "Ребёнок", None]


# ------------------------------------------------------------
# ТЕСТ 4: присваивание результата в столбец
# ------------------------------------------------------------
print("--- ТЕСТ 4: присваивание в столбец ---")
m2 = ["Имя", "Возраст";
      "Аня",  25;
      "Боб",  15]
m2[:, end+1] = case(m2[:, "Возраст"],
                    when < 18 then "Дитя",
                    else "Взрослый")
print(m2)
print("")
# ОЖИДАНИЕ:
# Имя   Возраст  None
# Аня   25       Взрослый
# Боб   15       Дитя


# ------------------------------------------------------------
# ТЕСТ 5: отрицательное условие
# ------------------------------------------------------------
print("--- ТЕСТ 5: when != ---")
r = case(m[:, "Отдел"],
         when != "IT" then "Не IT",
         else "IT")
print(r)
print("")
# ОЖИДАНИЕ:
# ["IT", "Не IT", "IT", "Не IT", "Не IT", "Не IT"]


# ------------------------------------------------------------
# ТЕСТ 6: несколько WHEN
# ------------------------------------------------------------
print("--- ТЕСТ 6: несколько when ---")
r = case(m[:, "Зарплата"],
         when < 70000 then "Низкая",
         when < 85000 then "Средняя",
         else "Высокая")
print(r)
print("")
# ОЖИДАНИЕ:
# ["Высокая", "Низкая", "Средняя", "Средняя", "Средняя", "Высокая"]
