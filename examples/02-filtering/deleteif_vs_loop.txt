# deleteif vs обратный цикл + delete

# Исходная таблица
m = ["Отдел", "Сотрудник", "Зарплата";
     "IT",    "Аня",       85000;
     "HR",    "Боб",       65000;
     "IT",    "Света",     95000;
     "Sales", "Гоша",      80000;
     "IT",    "Даша",      70000;
     "HR",    "Егор",      72000]
println(m)
# После:
# Отдел  Сотрудник  Зарплата
# IT     Аня        85000
# HR     Боб        65000
# IT     Света      95000
# Sales  Гоша       80000
# IT     Даша       70000
# HR     Егор       72000


# Способ 1: deleteif — одна строка
a = deleteif(m[:, "Отдел"] == "IT")
println(a)
# После:
# Отдел  Сотрудник  Зарплата
# HR     Боб        65000
# Sales  Гоша       80000
# HR     Егор       72000


# Способ 2: обратный цикл + delete
b = m
for i = lenrow(b) to 2 do 
    if b[i, "Отдел"] == "IT" then 
       b = delete(b[i, :])
    

println(b)
# После:
# Отдел  Сотрудник  Зарплата
# HR     Боб        65000
# Sales  Гоша       80000
# HR     Егор       72000


# Проверка: результаты совпадают
println(lenrow(a))
println(lenrow(b))
println(lenrow(a) == lenrow(b))
# После:
# 4
# 4
# True
