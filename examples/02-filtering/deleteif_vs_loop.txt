# ============================================================
# ДЕМО: deleteif  vs  обратный цикл + delete
# ============================================================
# Задача: УДАЛИТЬ сотрудников отдела "IT" (оставить всех остальных)
# ============================================================

# ------------------------------------------------------------
# 1. ИСХОДНЫЕ ДАННЫЕ
# ------------------------------------------------------------
m = ["Отдел", "Сотрудник", "Зарплата";
     "IT",    "Аня",       85000;
     "HR",    "Боб",       65000;
     "IT",    "Света",     95000;
     "Sales", "Гоша",      80000;
     "IT",    "Даша",      70000;
     "HR",    "Егор",      72000]

print("=== ИСХОДНАЯ ТАБЛИЦА ===")
print(m)

print()
print("=== СПОСОБ 1: deleteif с and/or ===")

# ------------------------------------------------------------
# 2. СПОСОБ 1 — deleteif (декларативно, в одну строку)
# ------------------------------------------------------------
print()
print("=== СПОСОБ 1: deleteif ===")

a = deleteif(m[:, "Отдел"] == "IT")
print(a)


# ------------------------------------------------------------
# 3. СПОСОБ 2 — ОБРАТНЫЙ цикл + delete
# ------------------------------------------------------------
print()
print("=== СПОСОБ 2: обратный цикл + delete ===")

# 3.1. Копируем исходную матрицу в b (чтобы не портить m)
b = m

# 3.2. Идём С КОНЦА ВВЕРХ: от последней строки до 2-й
#      Заголовок (строка 1) не трогаем!
for i = lenrow(b) to 2 do
{
    # 3.3. Если строка — "IT", удаляем её
    if b[i, "Отдел"] == "IT" then
    {
        # delete ВОЗВРАЩАЕТ новую матрицу → присваиваем в себя
        b = delete(b[i, :])
    }
}

print(b)


# ------------------------------------------------------------
# 4. ПРОВЕРКА: РЕЗУЛЬТАТЫ ДОЛЖНЫ СОВПАДАТЬ
# ------------------------------------------------------------
print()
print("=== ПРОВЕРКА ===")
print("Строк в a:", lenrow(a))
print("Строк в b:", lenrow(b))
print("Совпадают:", lenrow(a) == lenrow(b))
