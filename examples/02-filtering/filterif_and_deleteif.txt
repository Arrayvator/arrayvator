# filterif и deleteif дают один результат

# Исходная таблица
m = ["Отдел", "Сотрудник", "Зарплата";
     "IT",    "Аня",       85000;
     "HR",    "Боб",       65000;
     "IT",    "Света",     95000;
     "Sales", "Гоша",      80000;
     "IT",    "Даша",      70000;
     "HR",    "Егор",      72000]
println(m)
# После:
# Отдел  Сотрудник  Зарплата
# IT     Аня        85000
# HR     Боб        65000
# IT     Света      95000
# Sales  Гоша       80000
# IT     Даша       70000
# HR     Егор       72000


# Способ 1: filterif — оставить IT
a = filterif(m[:, "Отдел"] == "IT")
println(a)
# После:
# Отдел  Сотрудник  Зарплата
# IT     Аня        85000
# IT     Света      95000
# IT     Даша       70000


# Способ 2: deleteif — удалить НЕ IT
b = deleteif(not (m[:, "Отдел"] == "IT"))
println(b)
# После:
# Отдел  Сотрудник  Зарплата
# IT     Аня        85000
# IT     Света      95000
# IT     Даша       70000


# Проверка: совпадают
println(lenrow(a))
println(lenrow(b))
println(lenrow(a) == lenrow(b))
# После:
# 4
# 4
# True
