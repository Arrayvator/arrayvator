# ПРОВЕРКА 9: эквивалентность deleteif ↔ filterif(not ...) на BigData
# ============================================================
print()
print("=== 9. BigData: deleteif == filterif(not) ===")

eq_a = deleteif(bd[:, "Отдел"] == "IT" and bd[:, "Зарплата"] > 80000)
eq_b = filterif(not (bd[:, "Отдел"] == "IT" and bd[:, "Зарплата"] > 80000))

print("deleteif  — строк:", lenrow(eq_a) - 1)
print("filterif  — строк:", lenrow(eq_b) - 1)
print("Совпадают по количеству:", lenrow(eq_a) == lenrow(eq_b))

# Построчная проверка через ToMatrix
a_m = ToMatrix(eq_a)
b_m = ToMatrix(eq_b)

same = true
for i = 1 to lenrow(a_m) do
{
    if a_m[i, 1] != b_m[i, 1] or a_m[i, 2] != b_m[i, 2] or
       a_m[i, 3] != b_m[i, 3] or a_m[i, 4] != b_m[i, 4]
    then
    {
        same = false
        print("Различие в строке", i)
    }
}
print("Полное построчное совпадение:", same)
