# filterif: все варианты

# Исходная матрица
m = ["Имя", "Отдел", "Возраст", "Дата", "Зарплата";
     "Аня",   "IT",    25,       "20.06.2025", 85000;
     "Боб",   "HR",    32,       "15.07.2024", 65000;
     "Света", "IT",    28,       "10.08.2025", 95000;
     "Гена",  "Sales", 41,       "01.01.2024", 70000;
     "Даша",  "IT",    35,       "05.03.2025", 88000;
     "Егор",  "HR",    22,       "12.12.2024", 60000]
println(m)
# После:
# Имя    Отдел  Возраст  Дата        Зарплата
# Аня    IT     25       20.06.2025  85000
# Боб    HR     32       15.07.2024  65000
# Света  IT     28       10.08.2025  95000
# Гена   Sales  41       01.01.2024  70000
# Даша   IT     35       05.03.2025  88000
# Егор   HR     22       12.12.2024  60000


# filterif: Отдел == IT
r = filterif(m[:, "Отдел"] == "IT")
println(r)
# После:
# Имя    Отдел  Возраст  Дата        Зарплата
# Аня    IT     25       20.06.2025  85000
# Света  IT     28       10.08.2025  95000
# Даша   IT     35       05.03.2025  88000


# filterif: Возраст > 25
r = filterif(m[:, "Возраст"] > 25)
println(r)
# После:
# Имя    Отдел  Возраст  Дата        Зарплата
# Боб    HR     32       15.07.2024  65000
# Света  IT     28       10.08.2025  95000
# Гена   Sales  41       01.01.2024  70000
# Даша   IT     35       05.03.2025  88000


# filterif: IT AND Возраст > 25
r = filterif(m[:, "Отдел"] == "IT" and m[:, "Возраст"] > 25)
println(r)
# После:
# Имя    Отдел  Возраст  Дата        Зарплата
# Света  IT     28       10.08.2025  95000
# Даша   IT     35       05.03.2025  88000


# filterif: IT OR HR
r = filterif(m[:, "Отдел"] == "IT" or m[:, "Отдел"] == "HR")
println(r)
# После:
# Имя    Отдел  Возраст  Дата        Зарплата
# Аня    IT     25       20.06.2025  85000
# Боб    HR     32       15.07.2024  65000
# Света  IT     28       10.08.2025  95000
# Даша   IT     35       05.03.2025  88000
# Егор   HR     22       12.12.2024  60000


# filterif: NOT IT
r = filterif(not (m[:, "Отдел"] == "IT"))
println(r)
# После:
# Имя    Отдел  Возраст  Дата        Зарплата
# Боб    HR     32       15.07.2024  65000
# Гена   Sales  41       01.01.2024  70000
# Егор   HR     22       12.12.2024  60000


# filterif: 25 <= Возраст <= 35
r = filterif(m[:, "Возраст"] >= 25 and m[:, "Возраст"] <= 35)
println(r)
# После:
# Имя    Отдел  Возраст  Дата        Зарплата
# Аня    IT     25       20.06.2025  85000
# Боб    HR     32       15.07.2024  65000
# Света  IT     28       10.08.2025  95000
# Даша   IT     35       05.03.2025  88000


# filterif: год(Дата) == 2025 → ошибка (функция в условии)
try {
    r = filterif(year(m[:, "Дата"]) == 2025)
    println(r)
} catch {
    println(error)
}
# После:
# filterif: в условии используется функция year() — так нельзя.
#
#   filterif работает только с ГОТОВЫМ столбцом.
#   Он не умеет фильтровать по результату функции.
#
# РЕШЕНИЕ: сначала добавьте столбец через addcolumn,
# потом фильтруйте по нему.
# ... (полный текст подсказки)


# Обход: addcolumn + filterif по году
m2 = addcolumn(m, "Год", year(m[:, "Дата"]))
println(m2)
# После:
# Имя    Отдел  Возраст  Дата        Зарплата  Год
# Аня    IT     25       20.06.2025  85000     2025
# Боб    HR     32       15.07.2024  65000     2024
# Света  IT     28       10.08.2025  95000     2025
# Гена   Sales  41       01.01.2024  70000     2024
# Даша   IT     35       05.03.2025  88000     2025
# Егор   HR     22       12.12.2024  60000     2024


# filterif по добавленному столбцу Год
r = filterif(m2[:, "Год"] == 2025)
println(r)
# После:
# Имя    Отдел  Возраст  Дата        Зарплата  Год
# Аня    IT     25       20.06.2025  85000     2025
# Света  IT     28       10.08.2025  95000     2025
# Даша   IT     35       05.03.2025  88000     2025


# Обход: addcolumn + filterif по месяцу
m3 = addcolumn(m, "Месяц", month(m[:, "Дата"]))
r = filterif(m3[:, "Месяц"] == 6)
println(r)
# После:
# Имя    Отдел  Возраст  Дата        Зарплата  Месяц
# Аня    IT     25       20.06.2025  85000     6


# Обход: час через addcolumn + filterif
m4 = ["Имя", "Время";
      "Аня",  "14:30";
      "Боб",  "20:15";
      "Света","09:45";
      "Гена", "22:00"]
m5 = addcolumn(m4, "Час", hour(m4[:, "Время"]))
r = filterif(m5[:, "Час"] >= 18)
println(r)
# После:
# Имя    Время  Час
# Боб    20:15  20
# Гена   22:00  22


# Сложное: Отдел + Год
r = filterif(m2[:, "Отдел"] == "IT" and m2[:, "Год"] == 2025)
println(r)
