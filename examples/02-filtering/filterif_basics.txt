# ============================================================
# ТЕСТ FILTERIF — сложные условия
# ============================================================

m = ["Имя", "Отдел", "Возраст", "Дата", "Зарплата";
     "Аня",   "IT",    25,       "20.06.2025", 85000;
     "Боб",   "HR",    32,       "15.07.2024", 65000;
     "Света", "IT",    28,       "10.08.2025", 95000;
     "Гена",  "Sales", 41,       "01.01.2024", 70000;
     "Даша",  "IT",    35,       "05.03.2025", 88000;
     "Егор",  "HR",    22,       "12.12.2024", 60000]

print("Исходная матрица:")
print(m)
print("")


# ============================================================
# 1. ПРОСТЫЕ УСЛОВИЯ (должны работать)
# ============================================================
print("▶ 1. Простые условия")
print("")

r = filterif(m[:, "Отдел"] == "IT")
print("  filterif(Отдел == IT):")
print(r)
print("")

r = filterif(m[:, "Возраст"] > 25)
print("  filterif(Возраст > 25):")
print(r)
print("")


# ============================================================
# 2. СОСТАВНЫЕ УСЛОВИЯ (and / or)
# ============================================================
print("▶ 2. Составные условия")
print("")

r = filterif(m[:, "Отдел"] == "IT" and m[:, "Возраст"] > 25)
print("  filterif(Отдел == IT AND Возраст > 25):")
print(r)
print("")

r = filterif(m[:, "Отдел"] == "IT" or m[:, "Отдел"] == "HR")
print("  filterif(Отдел == IT OR Отдел == HR):")
print(r)
print("")


# ============================================================
# 3. ОТРИЦАНИЕ
# ============================================================
print("▶ 3. Отрицание")
print("")

r = filterif(not (m[:, "Отдел"] == "IT"))
print("  filterif(not (Отдел == IT)):")
print(r)
print("")


# ============================================================
# 4. ЧИСЛОВЫЕ ДИАПАЗОНЫ
# ============================================================
print("▶ 4. Числовые диапазоны")
print("")

r = filterif(m[:, "Возраст"] >= 25 and m[:, "Возраст"] <= 35)
print("  filterif(25 <= Возраст <= 35):")
print(r)
print("")


# ============================================================
# 5. ДИАПАЗОН СТРОК
# ============================================================
print("▶ 5. Диапазон строк")
print("")

r = filterif(m[2:5, "Отдел"] == "IT")
print("  filterif(m[2:5, Отдел] == IT):")
print(r)
print("")


# ============================================================
# 6. МОДИФИКАТОРЫ inside / ignore
# ============================================================
print("▶ 6. Модификаторы inside / ignore")
print("")

r = filterif(m[:, "Имя"] == "а", inside)
print("  filterif(Имя содержит 'а', inside):")
print(r)
print("")

r = filterif(m[:, "Имя"] == "аня", ignore)
print("  filterif(Имя == 'аня', ignore):")
print(r)
print("")


# ============================================================
# 7. ФУНКЦИЯ В УСЛОВИИ — ДОЛЖНА БЫТЬ ПОНЯТНАЯ ОШИБКА
# ============================================================
print("▶ 7. Функция в условии (ожидаем подсказку)")
print("")

try {
    r = filterif(year(m[:, "Дата"]) == 2025)
    print("  ⚠️  неожиданно сработало:", r)
}
catch {
    print("  ✅ Получили ожидаемую подсказку")
}
print("")


# ============================================================
# 8. ОБХОДНОЙ ПУТЬ: addcolumn + filterif
# ============================================================
print("▶ 8. Обходной путь через addcolumn")
print("")

m2 = addcolumn(m, "Год", year(m[:, "Дата"]))
print("  После addcolumn(m, 'Год', year(m[:, 'Дата'])):")
print(m2)
print("")

r = filterif(m2[:, "Год"] == 2025)
print("  filterif(m2[:, 'Год'] == 2025):")
print(r)
print("")


# ============================================================
# 9. ЕЩЁ ОДИН ОБХОДНОЙ ПУТЬ: месяц
# ============================================================
print("▶ 9. Обходной путь: месяц")
print("")

m3 = addcolumn(m, "Месяц", month(m[:, "Дата"]))
r = filterif(m3[:, "Месяц"] == 6)
print("  filterif(Месяц == 6):")
print(r)
print("")


# ============================================================
# 10. ОБХОДНОЙ ПУТЬ: час через время
# ============================================================
print("▶ 10. Обходной путь: час")
print("")

m4 = ["Имя", "Время";
      "Аня",  "14:30";
      "Боб",  "20:15";
      "Света","09:45";
      "Гена", "22:00"]

m5 = addcolumn(m4, "Час", hour(m4[:, "Время"]))
r = filterif(m5[:, "Час"] >= 18)
print("  filterif(Час >= 18):")
print(r)
print("")


# ============================================================
# 11. СЛОЖНОЕ УСЛОВИЕ: базовый столбец + обходной
# ============================================================
print("▶ 11. Сложное: Отдел + Год")
print("")

r = filterif(m2[:, "Отдел"] == "IT" and m2[:, "Год"] == 2025)
print("  filterif(Отдел == IT AND Год == 2025):")
print(r)
print("")
