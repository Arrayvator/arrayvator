# filterif vs цикл

# Исходная матрица
m = [1, "Имя", "Отдел", "Возраст", "Зарплата";
     2,   "Аня",   "IT",    25,       85000;
     3,   "Боб",   "HR",    32,       65000;
     4,   "Света", "IT",    28,       95000;
     5,   "Гена",  "Sales", 41,       70000;
     6,   "Даша",  "IT",    35,       88000;
     7,   "Егор",  "HR",    22,       60000]
println(m)

# После:
# 1  Имя    Отдел  Возраст  Зарплата
# 2  Аня    IT     25       85000
# 3  Боб    HR     32       65000
# 4  Света  IT     28       95000
# 5  Гена   Sales  41       70000
# 6  Даша   IT     35       88000
# 7  Егор   HR     22       60000

println()

# Способ 1: filterif — оставить IT
r = filterif(m[:, "Отдел"] == "IT")
println(r)

println()
