# ============================================================
# ДЕМО: filterif  vs  обратный цикл + delete
# ============================================================
# Задача: оставить только сотрудников отдела "IT"
# ============================================================

# ------------------------------------------------------------
# 1. ИСХОДНЫЕ ДАННЫЕ
# ------------------------------------------------------------
m = ["Отдел", "Сотрудник", "Зарплата";
     "IT",    "Аня",       85000;
     "HR",    "Боб",       65000;
     "IT",    "Света",     95000;
     "Sales", "Гоша",      80000;
     "IT",    "Даша",      70000;
     "HR",    "Егор",      72000]

print("=== ИСХОДНАЯ ТАБЛИЦА ===")
print(m)


# ------------------------------------------------------------
# 2. СПОСОБ 1 — filterif (декларативно, в одну строку)
# ------------------------------------------------------------
print()
print("=== СПОСОБ 1: filterif ===")

a = filterif(m[:, "Отдел"] == "IT")
print(a)


# ------------------------------------------------------------
# 3. СПОСОБ 2 — ОБРАТНЫЙ цикл + delete
# ------------------------------------------------------------
print()
print("=== СПОСОБ 2: обратный цикл + delete ===")

# 3.1. Копируем матрицу (присваивание делает копию)
b = m

# 3.2. Идём С КОНЦА ВВЕРХ: от последней строки до 2-й
#      Заголовок (строка 1) не трогаем!
for i = lenrow(b) to 2 do
{
    # 3.3. Если строка НЕ "IT" — удаляем её
    if b[i, "Отдел"] != "IT" then
    {
        b = delete(b[i, :])
    }
}

print(b)


# ------------------------------------------------------------
# 4. ПРОВЕРКА: РЕЗУЛЬТАТЫ ДОЛЖНЫ СОВПАДАТЬ
# ------------------------------------------------------------
print()
print("=== ПРОВЕРКА ===")
print("Строк в a:", lenrow(a))
print("Строк в b:", lenrow(b))
print("Совпадают:", lenrow(a) == lenrow(b))
