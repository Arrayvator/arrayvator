m = ["Имя", "Отдел", "Возраст", "Дата", "Зарплата";
     "Аня",   "IT",    25,       "20.06.2025", 85000;
     "Боб",   "HR",    32,       "15.07.2024", 65000;
     "Света", "IT",    28,       "10.08.2025", 95000;
     "Гена",  "Sales", 41,       "01.01.2024", 70000;
     "Даша",  "IT",    35,       "05.03.2025", 88000;
     "Егор",  "HR",    22,       "12.12.2024", 60000]

print("Исходная матрица:")
print(m)
print("")


# ============================================================
# 1. ПРОСТЫЕ УСЛОВИЯ
# ============================================================
print("▶ 1. Простые условия")
print("")

r = deleteif(m[:, "Отдел"] == "IT")
print("  deleteif(Отдел == IT) — удалить IT:")
print(r)
print("")

r = deleteif(m[:] > 30)
print("  deleteif(Возраст > 30):")
print(r)
print("")
