# filterif: диапазон строк

# Исходная матрица
m = [1, "Имя", "Отдел", "Возраст", "Дата", "Зарплата";
     2,   "Аня",   "IT",    25,       "20.06.2025", 85000;
     3,   "Боб",   "HR",    32,       "15.07.2024", 65000;
     4,   "Света", "IT",    28,       "10.08.2025", 95000;
     5,   "Гена",  "Sales", 41,       "01.01.2024", 70000;
     6,   "Даша",  "IT",    35,       "05.03.2025", 88000;
     7,   "Егор",  "HR",    22,       "12.12.2024", 60000]
println(m)
# После:
# 1  Имя    Отдел  Возраст  Дата        Зарплата
# 2  Аня    IT     25       20.06.2025  85000
# 3  Боб    HR     32       15.07.2024  65000
# 4  Света  IT     28       10.08.2025  95000
# 5  Гена   Sales  41       01.01.2024  70000
# 6  Даша   IT     35       05.03.2025  88000
# 7  Егор   HR     22       12.12.2024  60000

println()
# filterif: только строки 2–5, условие IT
r = filterif(m[2:5, "Отдел"] == "IT")
println(r)
