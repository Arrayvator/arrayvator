# Сортировка

# Исходная матрица
m = ["Имя", "Отдел", "Возраст", "Зарплата";
     "Света", "IT",    28,      95000;
     "Аня",   "HR",    25,      65000;
     "Боб",   "IT",    32,      85000;
     "Гена",  "Sales", 41,      70000;
     "Даша",  "HR",    35,      88000;
     "Егор",  "IT",    22,      60000]
println(m)
# После:
# Имя    Отдел  Возраст  Зарплата
# Света  IT     28       95000
# Аня    HR     25       65000
# Боб    IT     32       85000
# Гена   Sales  41       70000
# Даша   HR     35       88000
# Егор   IT     22       60000


# sort: Имя AZ (по алфавиту А→Я)
r = sort(m[:, "Имя"], AZ)
println(r)
# После:
# Имя    Отдел  Возраст  Зарплата
# Аня    HR     25       65000
# Боб    IT     32       85000
# Гена   Sales  41       70000
# Даша   HR     35       88000
# Егор   IT     22       60000
# Света  IT     28       95000


# sort: Имя ZA (по алфавиту Я→А)
r = sort(m[:, "Имя"], ZA)
println(r)
# После:
# Имя    Отдел  Возраст  Зарплата
# Света  IT     28       95000
# Егор   IT     22       60000
# Даша   HR     35       88000
# Гена   Sales  41       70000
# Боб    IT     32       85000
# Аня    HR     25       65000


# sort: Возраст AZ (по возрастанию)
r = sort(m[:, "Возраст"], AZ)
println(r)
# После:
# Имя    Отдел  Возраст  Зарплата
# Егор   IT     22       60000
# Аня    HR     25       65000
# Света  IT     28       95000
# Боб    IT     32       85000
# Даша   HR     35       88000
# Гена   Sales  41       70000


# sort: Возраст ZA (по убыванию)
r = sort(m[:, "Возраст"], ZA)
println(r)

# Мутация: m2 = sort(m2[:, "Возраст"], AZ)

# sort + filterif: IT, потом по возрасту
r1 = filterif(m[:, "Отдел"] == "IT")
println(r1)
# После:
# Имя    Отдел  Возраст  Зарплата

# Боб    IT     32       85000
# Егор   IT     22       60000


r2 = sort(r1[:, "Возраст"], AZ)
println(r2)
# После:
# Имя    Отдел  Возраст  Зарплата
# Егор   IT     22       60000
# Света  IT     28       95000
# Боб    IT     32       85000


# sort + addcolumn: год
m3 = ["Имя", "Дата";
      "Света", "10.08.2025";
      "Аня",   "20.06.2025";
      "Боб",   "15.07.2024"]
m4 = addcolumn(m3, "Год", year(m3[:, "Дата"]))
println(m4)
# После:
# Имя    Дата        Год
# Света  10.08.2025  2025
# Аня    20.06.2025  2025
# Боб    15.07.2024  2024


r = sort(m4[:, "Год"], AZ)
println(r)
# После:
# Имя    Дата        Год
# Боб    15.07.2024  2024
# Света  10.08.2025  2025
# Аня    20.06.2025  2025
