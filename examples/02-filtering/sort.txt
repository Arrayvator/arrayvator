# ============================================================
# ТЕСТ SORT — сортировка с матрицей
# ============================================================
# Проверяем:
#   1. Простые сортировки (AZ / ZA)
#   2. Сортировка по числам и строкам
#   3. Сортировка срезов (диапазон строк)
#   4. Сортировка вектора
#   5. Сортировка с None
#   6. Мутация (m = sort(...))
# ============================================================

print("=")
print("ТЕСТ SORT — СОРТИРОВКА")
print("=")
print("")


# ============================================================
# ДАННЫЕ
# ============================================================
m = ["Имя", "Отдел", "Возраст", "Зарплата";
     "Света", "IT",    28,      95000;
     "Аня",   "HR",    25,      65000;
     "Боб",   "IT",    32,      85000;
     "Гена",  "Sales", 41,      70000;
     "Даша",  "HR",    35,      88000;
     "Егор",  "IT",    22,      60000]

print("Исходная матрица:")
print(m)
print("")


# ============================================================
# 1. СОРТИРОВКА ПО СТРОКАМ (AZ / ZA)
# ============================================================
print("▶ 1. Сортировка по строкам (Имя)")
print("")

r = sort(m[:, "Имя"], AZ)
print("  sort(Имя, AZ) — по алфавиту А→Я:")
print(r)
print("")

r = sort(m[:, "Имя"], ZA)
print("  sort(Имя, ZA) — по алфавиту Я→А:")
print(r)
print("")


# ============================================================
# 2. СОРТИРОВКА ПО ЧИСЛАМ
# ============================================================
print("▶ 2. Сортировка по числам (Возраст)")
print("")

r = sort(m[:, "Возраст"], AZ)
print("  sort(Возраст, AZ) — по возрастанию:")
print(r)
print("")

r = sort(m[:, "Возраст"], ZA)
print("  sort(Возраст, ZA) — по убыванию:")
print(r)
print("")


# ============================================================
# 3. СОРТИРОВКА ПО ЗАРПЛАТЕ (числа, ZA)
# ============================================================
print("▶ 3. Топ зарплат")
print("")

r = sort(m[:, "Зарплата"], ZA)
print("  sort(Зарплата, ZA) — самые большие сверху:")
print(r)
print("")


# ============================================================
# 4. СОРТИРОВКА СРЕЗА СТРОК
# ============================================================
print("▶ 4. Сортировка только части строк (2:4)")
print("")

r = sort(m[2:4, "Возраст"], AZ)
print("  sort(m[2:4, Возраст], AZ) — сортируем только строки 2-4:")
print(r)
print("")


# ============================================================
# 5. СОРТИРОВКА ВЕКТОРА
# ============================================================
print("▶ 5. Сортировка вектора")
print("")

v = [30, 10, 50, 20, 40]
print("  Исходный вектор:", v)

r = sort(v, AZ)
print("  sort(v, AZ):", r)

r = sort(v, ZA)
print("  sort(v, ZA):", r)
print("")


# ============================================================
# 6. СОРТИРОВКА С ПУСТЫМИ ЗНАЧЕНИЯМИ (None)
# ============================================================
print("▶ 6. Сортировка с None")
print("")

m_none = ["Имя", "Зарплата";
          "Света", 95000;
          "Аня",   None;
          "Боб",   85000;
          "Гена",  None;
          "Даша",  88000]

print("  Исходная матрица с None:")
print(m_none)
print("")

r = sort(m_none[:, "Зарплата"], AZ)
print("  sort(Зарплата, AZ) — None уходят в конец:")
print(r)
print("")

r = sort(m_none[:, "Зарплата"], ZA)
print("  sort(Зарплата, ZA) — None уходят в конец:")
print(r)
print("")


# ============================================================
# 7. МУТАЦИЯ — сортировка матрицы на месте
# ============================================================
print("▶ 7. Мутация (m = sort(...))")
print("")

m2 = ["Имя", "Возраст";
      "Света", 28;
      "Аня",   25;
      "Боб",   32]

print("  До мутации:")
print(m2)
print("")

m2 = sort(m2[:, "Возраст"], AZ)
print("  После m2 = sort(m2[:, 'Возраст'], AZ):")
print(m2)
print("")


# ============================================================
# 8. КОМБИНАЦИЯ: sort + filterif
# ============================================================
print("▶ 8. sort + filterif")
print("")

r1 = filterif(m[:, "Отдел"] == "IT")
print("  1. filterif(Отдел == IT):")
print(r1)
print("")

r2 = sort(r1[:, "Возраст"], AZ)
print("  2. sort(Возраст, AZ):")
print(r2)
print("")


# ============================================================
# 9. КОМБИНАЦИЯ: sort + addcolumn
# ============================================================
print("▶ 9. sort + addcolumn (год)")
print("")

m3 = ["Имя", "Дата";
      "Света", "10.08.2025";
      "Аня",   "20.06.2025";
      "Боб",   "15.07.2024"]

m4 = addcolumn(m3, "Год", year(m3[:, "Дата"]))
print("  После addcolumn(Год):")
print(m4)
print("")

r = sort(m4[:, "Год"], AZ)
print("  sort(Год, AZ):")
print(r)
print("")
