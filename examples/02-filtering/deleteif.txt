# Фильтрация: deleteif

# Исходная матрица
m = ["Имя", "Отдел", "Возраст", "Дата", "Зарплата";
     "Аня",   "IT",    25,       "20.06.2025", 85000;
     "Боб",   "HR",    32,       "15.07.2024", 65000;
     "Света", "IT",    28,       "10.08.2025", 95000;
     "Гена",  "Sales", 41,       "01.01.2024", 70000;
     "Даша",  "IT",    35,       "05.03.2025", 88000;
     "Егор",  "HR",    22,       "12.12.2024", 60000]
println(m)
# После:
# Имя    Отдел  Возраст  Дата        Зарплата
# Аня    IT     25       20.06.2025  85000
# Боб    HR     32       15.07.2024  65000
# Света  IT     28       10.08.2025  95000
# Гена   Sales  41       01.01.2024  70000
# Даша   IT     35       05.03.2025  88000
# Егор   HR     22       12.12.2024  60000


# deleteif: удалить IT
r = deleteif(m[:, "Отдел"] == "IT")
println(r)
# После:
# Имя    Отдел  Возраст  Дата        Зарплата
# Боб    HR     32       15.07.2024  65000
# Гена   Sales  41       01.01.2024  70000
# Егор   HR     22       12.12.2024  60000


# deleteif: удалить возраст > 30
r = deleteif(m[:, "Возраст"] > 30)
println(r)
# После:
# Имя    Отдел  Возраст  Дата        Зарплата
# Аня    IT     25       20.06.2025  85000
# Света  IT     28       10.08.2025  95000
# Егор   HR     22       12.12.2024  60000


# deleteif: IT AND зарплата > 90000
r = deleteif(m[:, "Отдел"] == "IT" and m[:, "Зарплата"] > 90000)
println(r)
# После:
# Имя    Отдел  Возраст  Дата        Зарплата
# Аня    IT     25       20.06.2025  85000
# Боб    HR     32       15.07.2024  65000
# Гена   Sales  41       01.01.2024  70000
# Даша   IT     35       05.03.2025  88000
# Егор   HR     22       12.12.2024  60000


# deleteif: IT OR HR
r = deleteif(m[:, "Отдел"] == "IT" or m[:, "Отдел"] == "HR")
println(r)
# После:
# Имя    Отдел  Возраст  Дата        Зарплата
# Гена   Sales  41       01.01.2024  70000


# deleteif: NOT IT
r = deleteif(not (m[:, "Отдел"] == "IT"))
println(r)
# После:
# Имя    Отдел  Возраст  Дата        Зарплата
# Аня    IT     25       20.06.2025  85000
# Света  IT     28       10.08.2025  95000
# Даша   IT     35       05.03.2025  88000


# deleteif: (IT AND >90000) OR (HR AND >40)
r = deleteif(
    (m[:, "Отдел"] == "IT" and m[:, "Зарплата"] > 90000) or
    (m[:, "Отдел"] == "HR" and m[:, "Возраст"] > 40)
)
println(r)
# После:
# Имя    Отдел  Возраст  Дата        Зарплата
# Аня    IT     25       20.06.2025  85000
# Боб    HR     32       15.07.2024  65000
# Гена   Sales  41       01.01.2024  70000
# Даша   IT     35       05.03.2025  88000
# Егор   HR     22       12.12.2024  60000


# deleteif: Sales OR возраст < 25
r = deleteif(m[:, "Отдел"] == "Sales" or m[:, "Возраст"] < 25)
println(r)
# После:
# Имя    Отдел  Возраст  Дата        Зарплата
# Боб    HR     32       15.07.2024  65000
# Света  IT     28       10.08.2025  95000
# Даша   IT     35       05.03.2025  88000


# deleteif: NOT (IT OR HR)
r = deleteif(not (m[:, "Отдел"] == "IT" or m[:, "Отдел"] == "HR"))
println(r)
# После:
# Имя    Отдел  Возраст  Дата        Зарплата
# Аня    IT     25       20.06.2025  85000
# Боб    HR     32       15.07.2024  65000
# Света  IT     28       10.08.2025  95000
# Даша   IT     35       05.03.2025  88000
# Егор   HR     22       12.12.2024  60000
