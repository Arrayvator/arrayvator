# ============================================================
# ФУНКЦИЯ: denserank
# ============================================================
# ЧТО ДЕЛАЕТ:
#   Присваивает строке ПЛОТНЫЙ ранг внутри группы.
#   При одинаковых значениях — одинаковый ранг,
#   НО следующий ранг НЕ ПРОПУСКАЕТСЯ (1, 2, 2, 3).
#
# ЗАЧЕМ НУЖНА:
#   Когда важно "сколько уникальных уровней" в группе,
#   а не "какое место по списку".
#   Например: категории товаров по цене.
#
# ОТЛИЧИЕ ОТ rank:
#   rank      → 1, 2, 2, 4 (пропуск после дубля)
#   denserank → 1, 2, 2, 3 (без пропуска)
#
# СИНТАКСИС:
#   denserank(m [, by m[:, "X"]] [, order m[:, "Y"]] [, AZ | ZA])
#
# СТОЛБЕЦ РЕЗУЛЬТАТА: __denserank
# ============================================================

m = ["Отдел", "Сотрудник", "Зарплата", "Возраст", "Год";
     "IT",      "Аня",       85000,      25,       2023;
     "IT",      "Боб",       95000,      32,       2023;
     "IT",      "Света",     70000,      28,       2024;
     "HR",      "Гена",      65000,      41,       2023;
     "HR",      "Даша",      72000,      35,       2024;
     "Sales",   "Егор",      80000,      29,       2023;
     "Sales",   "Жанна",     88000,      38,       2024;
     "Sales",   "Зина",      60000,      22,       2024]


# ------------------------------------------------------------
# 1. База
# ------------------------------------------------------------
print("--- 1. denserank(by Отдел, order Зарплата, ZA) ---")
r = denserank(m, by m[:, "Отдел"], order m[:, "Зарплата"], ZA)
print(r)
print("")
# ОЖИДАНИЕ: то же, что rank (нет дубликатов): IT: 2,1,3 / HR: 2,1 / Sales: 2,1,3


# ------------------------------------------------------------
# 2. Без by
# ------------------------------------------------------------
print("--- 2. denserank(order Зарплата, AZ) — без by ---")
r = denserank(m, order m[:, "Зарплата"], AZ)
print(r)
print("")
# ОЖИДАНИЕ: 1..8 (нет дубликатов)


# ------------------------------------------------------------
# 3. Без order
# ------------------------------------------------------------
print("--- 3. denserank(by Отдел) — без order ---")
r = denserank(m, by m[:, "Отдел"])
print(r)
print("")
# ОЖИДАНИЕ: IT: 1,2,3 / HR: 1,2 / Sales: 1,2,3


# ------------------------------------------------------------
# 4. ГЛАВНОЕ: дубликаты — без пропуска!
# ------------------------------------------------------------
print("--- 4. denserank с дубликатами: БЕЗ пропуска ---")
d = ["Имя", "Балл";
     "Аня",  100;
     "Боб",  100;
     "Света", 90;
     "Гена",  80]
r = denserank(d, order d[:, "Балл"], ZA)
print(r)
print("")
# ОЖИДАНИЕ: Аня=1, Боб=1, Света=2, Гена=3
# (сравни с rank: было 1, 1, 3, 4)
