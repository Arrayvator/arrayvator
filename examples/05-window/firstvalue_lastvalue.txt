# ============================================================
# ФУНКЦИИ: firstvalue и lastvalue
# ============================================================
# ЧТО ДЕЛАЮТ:
#   firstvalue — ПЕРВОЕ значение в группе (после сортировки).
#   lastvalue  — ПОСЛЕДНЕЕ значение в группе.
#   Значение подставляется ВСЕМ строкам группы.
#
# ЗАЧЕМ НУЖНЫ:
#   Показать "эталон" группы: "лучший результат отдела",
#   "худший показатель магазина" — для сравнения каждой
#   строки с эталоном.
#   Классика: анализ "выше/ниже среднего по группе".
#
# ОТЛИЧИЕ ОТ lag/lead:
#   lag/lead возвращают значение соседней строки.
#   firstvalue/lastvalue возвращают ОДНО значение для всей группы.
#
# СИНТАКСИС:
#   firstvalue(m[:, "X"] [, by ...] [, order ...] [, AZ | ZA])
#   lastvalue(m[:, "X"] [, by ...] [, order ...] [, AZ | ZA])
#
# СТОЛБЦЫ РЕЗУЛЬТАТА: __firstvalue, __lastvalue
# ============================================================

m = ["Отдел", "Сотрудник", "Зарплата", "Возраст", "Год";
     "IT",      "Аня",       85000,      25,       2023;
     "IT",      "Боб",       95000,      32,       2023;
     "IT",      "Света",     70000,      28,       2024;
     "HR",      "Гена",      65000,      41,       2023;
     "HR",      "Даша",      72000,      35,       2024;
     "Sales",   "Егор",      80000,      29,       2023;
     "Sales",   "Жанна",     88000,      38,       2024;
     "Sales",   "Зина",      60000,      22,       2024]


# ------------------------------------------------------------
# 1. FIRSTVALUE — максимум в группе (при ZA)
# ------------------------------------------------------------
print("--- 1. firstvalue(Зарплата, by Отдел, order Зарплата, ZA) ---")
r = firstvalue(m[:, "Зарплата"],
               by m[:, "Отдел"],
               order m[:, "Зарплата"],
               ZA)
print(r)
print("")
# ОЖИДАНИЕ: у всех IT = 95000, у HR = 72000, у Sales = 88000


# ------------------------------------------------------------
# 2. LASTVALUE — минимум в группе (при ZA)
# ------------------------------------------------------------
print("--- 2. lastvalue(Зарплата, by Отдел, order Зарплата, ZA) ---")
r = lastvalue(m[:, "Зарплата"],
              by m[:, "Отдел"],
              order m[:, "Зарплата"],
              ZA)
print(r)
print("")
# ОЖИДАНИЕ: у всех IT = 70000, у HR = 65000, у Sales = 60000


# ------------------------------------------------------------
# 3. FIRSTVALUE — минимум (при AZ)
# ------------------------------------------------------------
print("--- 3. firstvalue(Зарплата, by Отдел, order Зарплата, AZ) ---")
r = firstvalue(m[:, "Зарплата"],
               by m[:, "Отдел"],
               order m[:, "Зарплата"],
               AZ)
print(r)
print("")
# ОЖИДАНИЕ: у всех IT = 70000, у HR = 65000, у Sales = 60000


# ------------------------------------------------------------
# 4. LASTVALUE без by — единое значение для всей таблицы
# ------------------------------------------------------------
print("--- 4. lastvalue(Зарплата, order Зарплата, ZA) ---")
r = lastvalue(m[:, "Зарплата"],
              order m[:, "Зарплата"],
              ZA)
print(r)
print("")
# ОЖИДАНИЕ: у всех строк = 60000 (минимум по всей таблице)
