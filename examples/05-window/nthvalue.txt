# ============================================================
# ФУНКЦИЯ: nthvalue
# ============================================================
# ЧТО ДЕЛАЕТ:
#   Возвращает N-е значение в группе (после сортировки).
#   Подставляет его ВСЕМ строкам группы.
#
# ЗАЧЕМ НУЖНА:
#   Взять "серебро", "бронзу" группы, а не только первое/последнее.
#   Например: "показать 2-е место по продажам в отделе".
#
# ПАРАМЕТР N:
#   1 → то же, что firstvalue
#   N = размер группы → то же, что lastvalue
#   N больше размера → None
#
# СИНТАКСИС:
#   nthvalue(m[:, "X"], N [, by ...] [, order ...] [, AZ | ZA])
#
# СТОЛБЕЦ РЕЗУЛЬТАТА: __nthvalue
# ============================================================

m = ["Отдел", "Сотрудник", "Зарплата", "Возраст", "Год";
     "IT",      "Аня",       85000,      25,       2023;
     "IT",      "Боб",       95000,      32,       2023;
     "IT",      "Света",     70000,      28,       2024;
     "HR",      "Гена",      65000,      41,       2023;
     "HR",      "Даша",      72000,      35,       2024;
     "Sales",   "Егор",      80000,      29,       2023;
     "Sales",   "Жанна",     88000,      38,       2024;
     "Sales",   "Зина",      60000,      22,       2024]


# ------------------------------------------------------------
# 1. N=2: второе место по зарплате в отделе
# ------------------------------------------------------------
print("--- 1. nthvalue(Зарплата, 2, by Отдел, order Зарплата, ZA) ---")
r = nthvalue(m[:, "Зарплата"], 2,
             by m[:, "Отдел"],
             order m[:, "Зарплата"],
             ZA)
print(r)
print("")
# ОЖИДАНИЕ:
# IT (ZA: 95000, 85000, 70000) → 2-е = 85000 у всех IT
# HR (ZA: 72000, 65000)        → 2-е = 65000 у всех HR
# Sales (ZA: 88000, 80000, 60000) → 2-е = 80000 у всех Sales


# ------------------------------------------------------------
# 2. N=1: то же, что firstvalue
# ------------------------------------------------------------
print("--- 2. nthvalue(Зарплата, 1, by Отдел, order Зарплата, ZA) ---")
r = nthvalue(m[:, "Зарплата"], 1,
             by m[:, "Отдел"],
             order m[:, "Зарплата"],
             ZA)
print(r)
print("")
# ОЖИДАНИЕ: у всех IT = 95000, у HR = 72000, у Sales = 88000


# ------------------------------------------------------------
# 3. N=3: третье место
# ------------------------------------------------------------
print("--- 3. nthvalue(Зарплата, 3, by Отдел, order Зарплата, ZA) ---")
r = nthvalue(m[:, "Зарплата"], 3,
             by m[:, "Отдел"],
             order m[:, "Зарплата"],
             ZA)
print(r)
print("")
# ОЖИДАНИЕ:
# IT (3 значения)  → 3-е = 70000 у всех IT
# HR (2 значения)  → 3-го НЕТ → None у HR
# Sales (3 значения) → 3-е = 60000 у всех Sales


# ------------------------------------------------------------
# 4. БЕЗ by — Nth по всей таблице
# ------------------------------------------------------------
print("--- 4. nthvalue(Зарплата, 2, order Зарплата, ZA) ---")
r = nthvalue(m[:, "Зарплата"], 2,
             order m[:, "Зарплата"],
             ZA)
print(r)
print("")
# ОЖИДАНИЕ: 2-е по всей таблице = 88000 (Жанна) у всех строк
