# ============================================================
# ФУНКЦИЯ: rank
# ============================================================
# ЧТО ДЕЛАЕТ:
#   Присваивает строке ранг внутри группы (партиции).
#   Учитывает сортировку по указанному столбцу.
#
# ЗАЧЕМ НУЖНА:
#   Узнать, кто на каком месте внутри отдела/категории/группы.
#   Например: "топ-3 продавца в каждом магазине".
#
# ОТЛИЧИЕ ОТ rownumber:
#   При одинаковых значениях в order — rank даёт ОДИНАКОВЫЙ ранг
#   нескольким строкам и ПРОПУСКАЕТ следующий (1, 2, 2, 4).
#   rownumber всегда даёт 1, 2, 3, 4.
#
# СИНТАКСИС:
#   rank(m [, by m[:, "X"]] [, order m[:, "Y"]] [, AZ | ZA])
#
# СТОЛБЕЦ РЕЗУЛЬТАТА: __rank
# ============================================================

m = ["Отдел", "Сотрудник", "Зарплата", "Возраст", "Год";
     "IT",      "Аня",       85000,      25,       2023;
     "IT",      "Боб",       95000,      32,       2023;
     "IT",      "Света",     70000,      28,       2024;
     "HR",      "Гена",      65000,      41,       2023;
     "HR",      "Даша",      72000,      35,       2024;
     "Sales",   "Егор",      80000,      29,       2023;
     "Sales",   "Жанна",     88000,      38,       2024;
     "Sales",   "Зина",      60000,      22,       2024]


# ------------------------------------------------------------
# 1. БАЗОВЫЙ ВАРИАНТ: по отделам, по зарплате (ZA)
# ------------------------------------------------------------
print("--- 1. rank(by Отдел, order Зарплата, ZA) ---")
r = rank(m, by m[:, "Отдел"], order m[:, "Зарплата"], ZA)
print(r)
print("")


# ------------------------------------------------------------
# 2. БЕЗ by: одна группа — вся таблица
# ------------------------------------------------------------
print("--- 2. rank(order Зарплата, AZ) — без by ---")
r = rank(m, order m[:, "Зарплата"], AZ)
print(r)
print("")


# ------------------------------------------------------------
# 3. БЕЗ order: порядок строк как есть
# ------------------------------------------------------------
print("--- 3. rank(by Отдел) — без order ---")
r = rank(m, by m[:, "Отдел"])
print(r)
print("")


# ------------------------------------------------------------
# 4. БЕЗ by и без order: единый ранг по всей таблице
# ------------------------------------------------------------
print("--- 4. rank(m) — только таблица ---")
r = rank(m)
print(r)
print("")


# ------------------------------------------------------------
# 5. Проверка одинаковых значений (пропуск ранга)
# ------------------------------------------------------------
print("--- 5. rank с дубликатами: пропуск ранга ---")
d = ["Имя", "Балл";
     "Аня",  100;
     "Боб",  100;
     "Света", 90;
     "Гена",  80]
r = rank(d, order d[:, "Балл"], ZA)
print(r)
print("")
# ОЖИДАНИЕ: Аня=1, Боб=1, Света=3 (пропуск 2), Гена=4
