# ============================================================
# ГРУППА 2: ОКОННЫЕ ФУНКЦИИ
# percentrank / cumedist / winavg / wincount
# winmin / winmax / winmedian / winstdev / rownumber
# winsum / lag / lead
# ============================================================

m = ["Отдел", "Сотрудник", "Зарплата";
     "IT",    "Аня",       85000;
     "IT",    "Боб",       95000;
     "IT",    "Света",     70000;
     "HR",    "Гена",      65000;
     "HR",    "Даша",      72000;
     "Sales", "Егор",      80000;
     "Sales", "Жанна",     88000;
     "Sales", "Зина",      60000]

print("=== ИСХОДНАЯ ===")
print(m)
print("")


# ------------------------------------------------------------
# 1. percentrank (таблица + добавляет столбец)
# ------------------------------------------------------------
print("--- 1. percentrank(m, order Зарплата, AZ) ---")
r = percentrank(m, order m[:, "Зарплата"], AZ)
print(r)
print("")

print("--- 1b. percentrank(m, by Отдел, order Зарплата, ZA) ---")
r = percentrank(m, by m[:, "Отдел"], order m[:, "Зарплата"], ZA)
print(r)
print("")


# ------------------------------------------------------------

print("")

print("")

r = winavg(m[:, "Зарплата"], order m[:, "Зарплата"], AZ)

print("")
