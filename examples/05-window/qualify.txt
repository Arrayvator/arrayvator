# ============================================================
# ФУНКЦИЯ: qualify
# ============================================================
# ЧТО ДЕЛАЕТ:
#   Фильтрует строки по результату оконной функции.
#   Работает КАК WHERE, но для оконных функций.
#   Оконная функция вычисляется ДО фильтрации.
#
# ЗАЧЕМ НУЖНА:
#   Классика: "топ-3 по зарплате в каждом отделе".
#   Без qualify пришлось бы делать два шага:
#     1. rownumber(...) → новая таблица с столбцом
#     2. filterif(новая_таблица[:, "__rownumber"] <= 3)
#   С qualify — в одну строку.
#
# ПРАВИЛА:
#   - Первый аргумент — таблица (не срез).
#   - Второй — условие с оконной функцией внутри.
#   - ОП: <, >, <=, >=, ==, !=
#
# СИНТАКСИС:
#   qualify(m, <оконная_функция> <ОП> <значение>)
#
# РЕЗУЛЬТАТ:
#   Новая матрица с ТОЛЬКО прошедшими строками.
#   Столбцы исходной таблицы сохраняются.
# ============================================================

m = ["Отдел", "Сотрудник", "Зарплата", "Возраст", "Год";
     "IT",      "Аня",       85000,      25,       2023;
     "IT",      "Боб",       95000,      32,       2023;
     "IT",      "Света",     70000,      28,       2024;
     "HR",      "Гена",      65000,      41,       2023;
     "HR",      "Даша",      72000,      35,       2024;
     "Sales",   "Егор",      80000,      29,       2023;
     "Sales",   "Жанна",     88000,      38,       2024;
     "Sales",   "Зина",      60000,      22,       2024]


# ------------------------------------------------------------
# 1. Топ-2 по зарплате в каждом отделе
# ------------------------------------------------------------
print("--- 1. qualify: топ-2 по ЗП в отделе ---")
r = qualify(m, rownumber(m,
                         by m[:, "Отдел"],
                         order m[:, "Зарплата"],
                         ZA) <= 2)
print(r)
print("")
# ОЖИДАНИЕ (6 строк данных + заголовок = 7):
# IT: Боб (95000), Аня (85000)
# HR: Даша (72000), Гена (65000)
# Sales: Жанна (88000), Егор (80000)


# ------------------------------------------------------------
# 2. Топ-1 (только лучшие)
# ------------------------------------------------------------
print("--- 2. qualify: только лучший в отделе ---")
r = qualify(m, rank(m,
                    by m[:, "Отдел"],
                    order m[:, "Зарплата"],
                    ZA) == 1)
print(r)
print("")
# ОЖИДАНИЕ (3 строки + заголовок):
# IT: Боб (95000)
# HR: Даша (72000)
# Sales: Жанна (88000)


# ------------------------------------------------------------
# 3. Фильтр по winsum (накопительная сумма)
# ------------------------------------------------------------
print("--- 3. qualify: накопительная сумма <= 300000 ---")
r = qualify(m, winsum(m[:, "Зарплата"]) <= 300000)
print(r)
print("")
# ОЖИДАНИЕ: строки, где __winsum <= 300000
# 85000, 180000, 250000 → первые 3 строки (Аня, Боб, Света)


# ------------------------------------------------------------
# 4. Фильтр по ntile (первая квартиль)
# ------------------------------------------------------------
print("--- 4. qualify: только первая корзина ---")
r = qualify(m, ntile(m, 2, order m[:, "Зарплата"], AZ) == 1)
print(r)
print("")
# ОЖИДАНИЕ: 8 строк / 2 = 4, 4
# Первые 4 по возрастанию ЗП: Зина(60000), Гена(65000),
# Света(70000), Даша(72000)
