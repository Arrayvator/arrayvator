# ============================================================
# ФУНКЦИИ: lag и lead
# ============================================================
# ЧТО ДЕЛАЮТ:
#   lag  — берёт значение из ПРЕДЫДУЩЕЙ строки группы.
#   lead — берёт значение из СЛЕДУЮЩЕЙ строки группы.
#
# ЗАЧЕМ НУЖНЫ:
#   Сравнивать строки между собой: "насколько выросла зарплата
#   по сравнению с прошлым месяцем/годом".
#   Классика: анализ временных рядов, дельты, приросты.
#
# ПАРАМЕТРЫ:
#   offset  — на сколько строк назад/вперёд (по умолчанию 1).
#   default — что подставить, если вышли за пределы группы.
#             Обычно 0 или None.
#
# СИНТАКСИС:
#   lag(m[:, "X"] [, offset [, default]] [, by ...] [, order ...])
#   lead(m[:, "X"] [, offset [, default]] [, by ...] [, order ...])
#
# СТОЛБЦЫ РЕЗУЛЬТАТА: __lag, __lead
# ============================================================

m = ["Отдел", "Сотрудник", "Зарплата", "Возраст", "Год";
     "IT",      "Аня",       85000,      25,       2023;
     "IT",      "Боб",       95000,      32,       2023;
     "IT",      "Света",     70000,      28,       2024;
     "HR",      "Гена",      65000,      41,       2023;
     "HR",      "Даша",      72000,      35,       2024;
     "Sales",   "Егор",      80000,      29,       2023;
     "Sales",   "Жанна",     88000,      38,       2024;
     "Sales",   "Зина",      60000,      22,       2024]


# ------------------------------------------------------------
# 1. LAG базовый (offset=1, default=0)
# ------------------------------------------------------------
print("--- 1. lag(Зарплата, 1, 0) ---")
r = lag(m[:, "Зарплата"], 1, 0)
print(r)
print("")
# ОЖИДАНИЕ (вектор без заголовка):
# [0, 85000, 95000, 70000, 65000, 72000, 80000, 88000, 60000]
#  ^ первая = default, дальше — предыдущее


# ------------------------------------------------------------
# 2. LEAD базовый
# ------------------------------------------------------------
print("--- 2. lead(Зарплата, 1, 0) ---")
r = lead(m[:, "Зарплата"], 1, 0)
print(r)
print("")
# ОЖИДАНИЕ:
# [95000, 70000, 65000, 72000, 80000, 88000, 60000, 0, 0]?
# Первые 8 — следующее значение, последние = default


# ------------------------------------------------------------
# 3. LAG с offset=2
# ------------------------------------------------------------
print("--- 3. lag(Зарплата, 2, 0) ---")
r = lag(m[:, "Зарплата"], 2, 0)
print(r)
print("")
# ОЖИДАНИЕ: [0, 0, 85000, 95000, 70000, 65000, 72000, 80000, 88000]


# ------------------------------------------------------------
# 4. LAG с by (внутри группы)
# ------------------------------------------------------------
print("--- 4. lag(Зарплата, 1, 0, by Отдел, order Зарплата, AZ) ---")
r = lag(m[:, "Зарплата"], 1, 0,
        by m[:, "Отдел"],
        order m[:, "Зарплата"],
        AZ)
print(r)
print("")
# ОЖИДАНИЕ: внутри каждого отдела — предыдущее значение
# по возрастанию зарплаты:
# IT (AZ: 70000, 85000, 95000): 0, 70000, 85000
# HR (AZ: 65000, 72000):        0, 65000
# Sales (AZ: 60000, 80000, 88000): 0, 60000, 80000


# ------------------------------------------------------------
# 5. LEAD с by
# ------------------------------------------------------------
print("--- 5. lead(Зарплата, 1, 0, by Отдел, order Зарплата, ZA) ---")
r = lead(m[:, "Зарплата"], 1, 0,
         by m[:, "Отдел"],
         order m[:, "Зарплата"],
         ZA)
print(r)
print("")
# ОЖИДАНИЕ: внутри каждого отдела — следующее значение
# по убыванию зарплаты (ZA):
# IT (ZA: 95000, 85000, 70000): 85000, 70000, 0
# HR (ZA: 72000, 65000):        65000, 0
# Sales (ZA: 88000, 80000, 60000): 80000, 60000, 0
