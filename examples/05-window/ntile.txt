# ============================================================
# ФУНКЦИЯ: ntile
# ============================================================
# ЧТО ДЕЛАЕТ:
#   Разбивает группу на N примерно равных корзин.
#   Каждой строке присваивает номер корзины (1..N).
#
# ЗАЧЕМ НУЖНА:
#   Разделить данные на квартили, квинтили, децили.
#   Например: "разбить клиентов на 4 группы по объёму покупок".
#
# ОСОБЕННОСТИ:
#   Если строк не делится ровно на N — первым корзинам
#   достаётся на 1 строку больше.
#   Пример: 7 строк, N=3 → 3, 2, 2.
#
# СИНТАКСИС:
#   ntile(m, N [, by m[:, "X"]] [, order m[:, "Y"]] [, AZ | ZA])
#
# СТОЛБЕЦ РЕЗУЛЬТАТА: __ntile
# ============================================================

m = ["Отдел", "Сотрудник", "Зарплата", "Возраст", "Год";
     "IT",      "Аня",       85000,      25,       2023;
     "IT",      "Боб",       95000,      32,       2023;
     "IT",      "Света",     70000,      28,       2024;
     "HR",      "Гена",      65000,      41,       2023;
     "HR",      "Даша",      72000,      35,       2024;
     "Sales",   "Егор",      80000,      29,       2023;
     "Sales",   "Жанна",     88000,      38,       2024;
     "Sales",   "Зина",      60000,      22,       2024]


# ------------------------------------------------------------
# 1. N=2, по отделам
# ------------------------------------------------------------
print("--- 1. ntile(m, 2, by Отдел, order Зарплата, ZA) ---")
r = ntile(m, 2, by m[:, "Отдел"], order m[:, "Зарплата"], ZA)
print(r)
print("")
# ОЖИДАНИЕ (ZA — от больших к меньшим):
# IT (3 строки): 95000→1, 85000→1, 70000→2
# HR (2 строки): 72000→1, 65000→2
# Sales (3 строки): 88000→1, 80000→1, 60000→2


# ------------------------------------------------------------
# 2. N=3, по всей таблице
# ------------------------------------------------------------
print("--- 2. ntile(m, 3, order Зарплата, AZ) ---")
r = ntile(m, 3, order m[:, "Зарплата"], AZ)
print(r)
print("")
# ОЖИДАНИЕ (AZ: 60000, 65000, 70000, 72000, 80000, 85000, 88000, 95000):
# 8 строк / 3 корзины = 3, 3, 2
# 60000→1, 65000→1, 70000→1, 72000→2, 80000→2, 85000→2, 88000→3, 95000→3


# ------------------------------------------------------------
# 3. N=4 — квартили
# ------------------------------------------------------------
print("--- 3. ntile(m, 4, order Зарплата, AZ) — квартили ---")
r = ntile(m, 4, order m[:, "Зарплата"], AZ)
print(r)
print("")
# ОЖИДАНИЕ: 8/4 = 2, 2, 2, 2
# 60000→1, 65000→1, 70000→2, 72000→2, 80000→3, 85000→3, 88000→4, 95000→4


# ------------------------------------------------------------
# 4. Неравное деление: 7 строк, N=3
# ------------------------------------------------------------
print("--- 4. ntile с неравным делением (7 строк, N=3) ---")
d = ["Имя", "Балл";
     "A", 10;
     "B", 20;
     "C", 30;
     "D", 40;
     "E", 50;
     "F", 60;
     "G", 70]
r = ntile(d, 3, order d[:, "Балл"], AZ)
print(r)
print("")
# ОЖИДАНИЕ: 7/3 → 3, 2, 2
# A=1, B=1, C=1, D=2, E=2, F=3, G=3
