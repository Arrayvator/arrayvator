# ============================================================
# 18. ВЕКТОР — срез столбца
# ============================================================
print("▶ Проверка на векторах (срез столбца)")

m = ["Дата", "Время";
     "20.06.2025", "14:30:15";
     "21.07.2025", "15:45:30";
     "22.08.2025", "09:15:00"]

# Год из каждой даты
r = year(m[:, "Дата"])
print("  year(m[:, \"Дата\"]):", r)

# Месяц из каждой даты
r = month(m[:, "Дата"])
print("  month(m[:, \"Дата\"]):", r)

# Час из времени
r = hour(m[:, "Время"])
print("  hour(m[:, \"Время\"]):", r)

# Минуты
r = minute(m[:, "Время"])
print("  minute(m[:, \"Время\"]):", r)
print("")
