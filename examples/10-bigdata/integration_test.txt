# ============================================================
# 0. КОНВЕРТАЦИЯ
# ============================================================

println("  lencol(bd) =", lencol(bd))
println("  type(bd)   =", type(bd))

# ============================================================
# 1. ЧТЕНИЕ ИЗ BD
# ============================================================
println("[1] ЧТЕНИЕ ИЗ BD")

hdr = bd[0, :]
println("  bd[0, :]              — заголовки       ✅")

last_row = bd[end, :]

s_cnt = count(bd[:, "Зарплата"])

s_std = std(bd[:, "Зарплата"])

println("  sum     =", s_sum, "  ✅")

println("  max     =", s_max, "  ✅")
println("  avg     =", s_avg, "  ✅")

println("  std     =", s_std, "  ✅")
println("")

# 3. СТАТИСТИКА С by
