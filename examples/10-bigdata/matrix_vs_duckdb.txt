# ============================================================
# ЧАСТЬ B: УПАВШИЕ ТЕСТЫ BIGDATA
# ============================================================

m = ["ID", "Имя", "Отдел", "Город", "Дата", "Зарплата", "Сумма";
      1, "Аня",    "IT",    "Москва",  "15.01.2025", 85000, 5000;
      2, "Боб",    "HR",    "Питер",   "20.02.2025", 65000, 3000;
      3, "Света",  "IT",    "Москва",  "05.03.2025", 95000, 7000;
      4, "Гена",   "Sales", "Казань",  "10.04.2025", 80000, 4000;
      5, "Даша",   "IT",    "Питер",   "15.05.2025", 88000, 6000;
      6, "Егор",   "HR",    "Москва",  "20.06.2025", 70000, 3500;
      7, "Жанна",  "Sales", "Казань",  "25.07.2025", 82000, 4500;
      8, "Зина",   "IT",    "Москва",  "30.08.2025", 92000, 6500;
      9, "Иван",   "HR",    "Питер",   "05.09.2025", 68000, 3200;
     10, "Костя",  "Sales", "Москва",  "10.10.2025", 78000, 4200;
     11, "Лена",   "IT",    "Казань",  "15.11.2025", 90000, 6200;
     12, "Миша",   "HR",    "Москва",  "20.12.2025", 72000, 3800;
     13, "Нина",   "Sales", "Питер",   "25.01.2026", 81000, 4400;
     14, "Олег",   "IT",    "Москва",  "01.02.2026", 87000, 5500;
     15, "Петя",   "HR",    "Казань",  "06.03.2026", 66000, 3100]

bd = ToBigData(m)

# ------------------------------------------------------------
# B1. PIVOT — через ToMatrix (DuckDB не поддерживает)
# ------------------------------------------------------------
println("[B1] pivot через ToMatrix")
m_bd = ToMatrix(bd)
r = pivot(m_bd[:, "Отдел"], m_bd[:, "Зарплата"], sum)
println(r)

# ------------------------------------------------------------
# B2. JOIN
# ------------------------------------------------------------
println("[B2] join(bd, bd_d, on \"Отдел\")")
d = ["Отдел", "Менеджер";
     "IT",    "Петров";
     "HR",    "Сидоров";
     "Sales", "Иванов"]
bd_d = ToBigData(d)
r = join(bd, bd_d, on "Отдел")
println(r)

# ------------------------------------------------------------
# B3. DEDUPLICATE
# ------------------------------------------------------------
println("[B3] DEDUPLICATE")
r = Unique(bd[:, "Отдел"])
println(r)

println("CountDistinct =", CountDistinct(bd[:, "Отдел"]))

r = ValueCounts(bd[:, "Отдел"])
println(r)

r = DeleteDuplicate(bd[:, "Отдел"])
println(r)

# ------------------------------------------------------------
# B4. UNPIVOT — через ToMatrix
# ------------------------------------------------------------
println("[B4] unpivot через ToMatrix")
r = unpivot(m_bd[:, 6:7], by m_bd[:, "Имя"])
println(r)

# ------------------------------------------------------------
# B5. CONDITIONAL AGGREGATES
# ------------------------------------------------------------
println("[B5] Условные агрегаты")
println("sumif =", sumif(bd[:, "Отдел"] == "IT", bd[:, "Зарплата"]))
println("countif =", countif(bd[:, "Отдел"] == "IT"))
println("avgif =", avgif(bd[:, "Отдел"] == "IT", bd[:, "Зарплата"]))
println("minif =", minif(bd[:, "Отдел"] == "IT", bd[:, "Зарплата"]))
println("maxif =", maxif(bd[:, "Отдел"] == "IT", bd[:, "Зарплата"]))
println("medianif =", medianif(bd[:, "Отдел"] == "IT", bd[:, "Зарплата"]))
println("countuniqueif =", countuniqueif(bd[:, "Отдел"] == "IT", bd[:, "Город"]))
println("sumproduct =", sumproduct(bd[:, "Зарплата"], bd[:, "Сумма"]))

# ------------------------------------------------------------
# B6. ANALYTICS
# ------------------------------------------------------------
println("[B6] Аналитика")
r = abc(bd[:, "Зарплата"])
println(r)
r = abc(bd[:, "Зарплата"], %)
println(r)
r = percentof(bd[:, "Зарплата"])
println(r)
r = percentof(bd[:, "Зарплата"], by bd[:, "Отдел"])
println(r)
r = anomaly(bd[:, "Зарплата"])
println(r)
r = anomaly(bd[:, "Зарплата"], only)
println(r)

# ------------------------------------------------------------
# B7. WINDOW FUNCTIONS
# ------------------------------------------------------------
println("[B7] Оконные функции")
r = rownumber(bd, by bd[:, "Отдел"], order bd[:, "Зарплата"], ZA)
println(r)
r = rank(bd, by bd[:, "Отдел"], order bd[:, "Зарплата"], ZA)
println(r)
r = denserank(bd, by bd[:, "Отдел"], order bd[:, "Зарплата"], ZA)
println(r)
r = ntile(bd, 3, by bd[:, "Отдел"], order bd[:, "Зарплата"], ZA)
println(r)
r = lag(bd[:, "Зарплата"], 1, 0)
println(r)
r = lead(bd[:, "Зарплата"], 1, 0)
println(r)
r = winsum(bd[:, "Зарплата"], by bd[:, "Отдел"])
println(r)
r = winavg(bd[:, "Зарплата"], by bd[:, "Отдел"])
println(r)
r = qualify(bd, rownumber(bd, by bd[:, "Отдел"], order bd[:, "Зарплата"], ZA) <= 2)
println(r)

# ------------------------------------------------------------
# B8. ADDCOLUMN
# ------------------------------------------------------------
println("[B8] addcolumn")
r = addcolumn(bd, "Премия", bd[:, "Зарплата"] * 0.1)
println(r)
println("lencol =", lencol(r))

# ------------------------------------------------------------
# B9. ТИПЫ / NULL
# ------------------------------------------------------------
println("[B9] Типы и None")
println("type(bd) =", type(bd))
println("type(m) =", type(m))
println("isnone(None) =", isnone(None))
println("coalesce(None, None, 10) =", coalesce(None, None, 10))

# ------------------------------------------------------------
# B10. RAM-only функции — ожидаются понятные ошибки
# ------------------------------------------------------------
println("[B10] RAM-only функции")
r = copy(bd[:, 1], bd[:, 3], after)
println(r)
r = move(bd[:, 1], bd[:, 3], after)
println(r)
r = insert(bd[:, 3], after)
println(r)
r = insertif(bd[:, "Отдел"] == "IT", after)
println(r)
r = matrixmod(bd, delete, 2)
println(r)
r = filldown(bd[:, "Отдел"])
println(r)
r = groupagg(bd[:, "Отдел"], bd[:, "Зарплата"], sum)
println(r)
r = transpose(bd)
println(r)

# ------------------------------------------------------------
# B11. ToMatrix
# ------------------------------------------------------------
println("[B11] ToMatrix")
m2 = ToMatrix(bd)
println(m2)
m3 = ToMatrix(bd, 5)
println(m3)
println("lenrow(m3) =", lenrow(m3))
