m = ["Товар",   "Категория", "Продажи";
     "Молоко",  "Молочка",   800;
     "Сыр",     "Молочка",   200;
     "Батон",   "Хлеб",      500]

r = percentof(m[:, "Продажи"])
print(r)
# ОЖИДАНИЕ: [53.33, 13.33, 33.33]
# (сумма 1500; 800/1500=53.33, 200/1500=13.33, 500/1500=33.33)

r2 = percentof(m[:, "Продажи"], by m[:, "Категория"])
print(r2)
# ОЖИДАНИЕ: [80.0, 20.0, 100.0]
# (Молочка: 800/1000=80, 200/1000=20; Хлеб: 500/500=100)
