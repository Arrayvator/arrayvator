# ============================================================
# ТЕСТ: Unique / DeleteDuplicate
# ============================================================

m = ["Отдел", "Имя", "Город";
     "IT",    "Аня", "Москва";
     "IT",    "Боб", "Питер";
     "HR",    "Света", "Москва";
     "IT",    "Гена", "Питер";
     "HR",    "Даша", "Москва";
     "Sales", "Егор", "Сочи"]

print("=== Исходная ===")
print(m)

# ============================================================
# 1. Unique — по столбцу "Отдел" (уникальные СТРОКИ)
# ============================================================
print("")
print("--- 1. Unique(m[:, 'Отдел']) ---")
r1 = Unique(m[:, "Отдел"])
print(r1)

# Ожидается: 3 строки — первое появление каждого отдела
#   Отдел  Имя    Город
#   IT     Аня    Москва
#   HR     Света  Москва
#   Sales  Егор   Сочи

# ============================================================
# 2. Unique — по столбцу "Город"
# ============================================================
print("")
print("--- 2. Unique(m[:, 'Город']) ---")
r2 = Unique(m[:, "Город"])
print(r2)

# Ожидается: 3 строки — первое появление каждого города
#   Отдел  Имя    Город
#   IT     Аня    Москва
#   IT     Боб    Питер
#   Sales  Егор   Сочи

# ============================================================
# 3. Unique — по вектору
# ============================================================
print("")
print("--- 3. Unique по вектору ---")
v = [10, 20, 20, 30, 30, 30, 40, 10]
r3 = Unique(v)
print(r3)

# Ожидается: [10, 20, 30, 40]

# ============================================================
# 4. Unique — по вектору с None
# ============================================================
print("")
print("--- 4. Unique с None ---")
v2 = ["A", "B", None, "A", None, "C"]
r4 = Unique(v2)
print(r4)

# Ожидается: [A, B, None, C]

# ============================================================
# 5. Unique — вся матрица (без среза)
# ============================================================
print("")
print("--- 5. Unique(m) ---")
m2 = ["Отдел";
      "IT";
      "IT";
      "HR";
      "IT"]
r5 = Unique(m2)
print(r5)

# Ожидается: 3 строки (IT, HR) — уникальные целиком
#   Отдел
#   IT
#   HR

# ============================================================
# 6. DeleteDuplicate — синоним Unique (по столбцу)
# ============================================================
print("")
print("--- 6. DeleteDuplicate(m[:, 'Отдел']) ---")
r6 = DeleteDuplicate(m[:, "Отдел"])
print(r6)

# Ожидается: то же, что Unique
#   IT     Аня    Москва
#   HR     Света  Москва
#   Sales  Егор   Сочи

# ============================================================
# 7. DeleteDuplicate — по вектору
# ============================================================
print("")
print("--- 7. DeleteDuplicate по вектору ---")
v3 = [5, 5, 3, 3, 3, 7, 5]
r7 = DeleteDuplicate(v3)
print(r7)

# Ожидается: [5, 3, 7]

# ============================================================
# 8. Unique — все одинаковые
# ============================================================
print("")
print("--- 8. Unique: все одинаковые ---")
m3 = ["X"; "A"; "A"; "A"; "A"]
r8 = Unique(m3[:, "X"])
print(r8)

# Ожидается:
#   X
#   A

# ============================================================
# 9. Unique — все разные
# ============================================================
print("")
print("--- 9. Unique: все разные ---")
m4 = ["X"; "A"; "B"; "C"; "D"]
r9 = Unique(m4[:, "X"])
print(r9)

# Ожидается: A, B, C, D (все)

# ============================================================
# 10. Unique — с числами
# ============================================================
print("")
print("--- 10. Unique: числа ---")
m5 = ["Возраст";
      25;
      25;
      30;
      30;
      30;
      41;
      25]
r10 = Unique(m5[:, "Возраст"])
print(r10)

# Ожидается: 25, 30, 41

# ============================================================
# 11. Проверка заголовков у результата
# ============================================================
print("")
print("--- 11. Заголовки Unique(m[:, 'Отдел']) ---")
r11 = Unique(m[:, "Отдел"])
print(r11[1, :])

# Ожидается: Отдел, Имя, Город

# ============================================================
# 12. Проверка исходной матрицы (мутация не должна произойти)
# ============================================================
print("")
print("--- 12. Исходная не должна измениться ---")
r12 = Unique(m[:, "Отдел"])
print(m)

# Ожидается: исходная m как была (6 строк данных)

print("")
print("=== ТЕСТ ЗАВЕРШЁН ===")
