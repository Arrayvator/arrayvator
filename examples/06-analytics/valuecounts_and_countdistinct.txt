# ============================================================
# ТЕСТ: ValueCounts / CountDistinct
# ============================================================

m = ["Отдел", "Имя", "Город";
     "IT",    "Аня", "Москва";
     "IT",    "Боб", "Питер";
     "HR",    "Света", "Москва";
     "IT",    "Гена", "Питер";
     "HR",    "Даша", "Москва";
     "Sales", "Егор", "Сочи"]

print("=== Исходная ===")
print(m)

# ============================================================
# 1. ValueCounts — частотная таблица по отделу
# ============================================================
print("")
print("--- 1. ValueCounts(m[:, 'Отдел']) ---")
r1 = ValueCounts(m[:, "Отдел"])
print(r1)

# ============================================================
# 2. ValueCounts — по городу (сортировка по частоте DESC)
# ============================================================
print("")
print("--- 2. ValueCounts(m[:, 'Город']) ---")
r2 = ValueCounts(m[:, "Город"])
print(r2)

# ============================================================
# 3. ValueCounts — по вектору
# ============================================================
print("")
print("--- 3. ValueCounts по вектору ---")
v = [10, 20, 20, 30, 30, 30, 40]
r3 = ValueCounts(v)
print(r3)

# ============================================================
# 4. ValueCounts — по столбцу с None
# ============================================================
print("")
print("--- 4. ValueCounts с None ---")
m2 = ["X";
      "A";
      "B";
      None;
      "A";
      "A";
      None]
r4 = ValueCounts(m2[:, "X"])
print(r4)

# ============================================================
# 5. CountDistinct — по столбцу "Отдел"
# ============================================================
print("")
print("--- 5. CountDistinct(m[:, 'Отдел']) ---")
r5 = CountDistinct(m[:, "Отдел"])
print(r5)

# ============================================================
# 6. CountDistinct — по столбцу "Город"
# ============================================================
print("")
print("--- 6. CountDistinct(m[:, 'Город']) ---")
r6 = CountDistinct(m[:, "Город"])
print(r6)

# ============================================================
# 7. CountDistinct — по вектору
# ============================================================
print("")
print("--- 7. CountDistinct по вектору ---")
v2 = [1, 1, 2, 3, 3, 3]
r7 = CountDistinct(v2)
print(r7)

# ============================================================
# 8. CountDistinct — с None
# ============================================================
print("")
print("--- 8. CountDistinct с None ---")
m3 = ["X";
      "A";
      "A";
      None;
      "B";
      None]
r8 = CountDistinct(m3[:, "X"])
print(r8)

# ============================================================
# 9. CountDistinct по числовому столбцу
# ============================================================
print("")
print("--- 9. CountDistinct по числам ---")

      25;
      25;
      30;
      30;
      30;

# 10. ValueCounts — все одинаковые

print("--- 10. ValueCounts: все одинаковые ---")
