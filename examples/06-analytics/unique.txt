# ============================================================
# ТЕСТ DuckDB через Convert_Matrix_To_BigData
# ============================================================

# ------------------------------------------------------------
# 1. Создаём матрицу с повторяющимися отделами и городами
# ------------------------------------------------------------
m = ["Отдел", "Имя", "Город";
     "IT",    "Аня", "Москва";
     "IT",    "Боб", "Питер";
     "HR",    "Света", "Москва";
     "IT",    "Гена", "Питер";
     "HR",    "Даша", "Москва";
     "Sales", "Егор", "Сочи"]

print("=== Исходная (Matrix) ===")
print(m)

# ------------------------------------------------------------
# 2. Конвертируем в DuckDB
# ------------------------------------------------------------
bd = Convert_Matrix_To_BigData(m)

print("")
print("=== BigData (DuckDB) ===")
print(bd)

# ============================================================
# 3. Unique по столбцу "Отдел" — ожидаем 3 строки, первое появление
# ============================================================
print("")
print("--- 3. Unique(bd[:, 'Отдел']) ---")
r3 = Unique(bd[:, "Отдел"])
print(r3)
# Ожидается:
#   IT     Аня    Москва
#   HR     Света  Москва
#   Sales  Егор   Сочи

# ============================================================
# 4. Unique по столбцу "Город" — ожидаем 3 строки
# ============================================================
print("")
print("--- 4. Unique(bd[:, 'Город']) ---")
r4 = Unique(bd[:, "Город"])
print(r4)
# Ожидается:
#   IT     Аня    Москва
#   IT     Боб    Питер
#   Sales  Егор   Сочи

# ============================================================
# 5. DeleteDuplicate по "Отдел" — должен совпадать с Unique
# ============================================================
print("")
print("--- 5. DeleteDuplicate(bd[:, 'Отдел']) ---")
r5 = DeleteDuplicate(bd[:, "Отдел"])
print(r5)
# Ожидается: ТО ЖЕ, ЧТО И r3
#   IT     Аня    Москва
#   HR     Света  Москва
#   Sales  Егор   Сочи

# ============================================================
# 6. ValueCounts по "Отдел" — IT=3, HR=2, Sales=1
# ============================================================
print("")
print("--- 6. ValueCounts(bd[:, 'Отдел']) ---")
r6 = ValueCounts(bd[:, "Отдел"])
print(r6)
# Ожидается:
#   Значение  Частота
#   IT        3
#   HR        2
#   Sales     1

# ============================================================
# 7. CountDistinct по "Отдел" — 3
# ============================================================
print("")
print("--- 7. CountDistinct(bd[:, 'Отдел']) ---")
r7 = CountDistinct(bd[:, "Отдел"])
print(r7)
# Ожидается: 3

# ============================================================
# 8. CountDistinct по "Город" — 3
# ============================================================
print("")
print("--- 8. CountDistinct(bd[:, 'Город']) ---")
r8 = CountDistinct(bd[:, "Город"])
print(r8)
# Ожидается: 3

# ============================================================
# 9. ValueCounts по "Город" — Москва=3, Питер=2, Сочи=1
# ============================================================
print("")
print("--- 9. ValueCounts(bd[:, 'Город']) ---")
r9 = ValueCounts(bd[:, "Город"])
print(r9)
# Ожидается:
#   Значение  Частота
#   Москва    3
#   Питер     2
#   Сочи      1

# ============================================================
# 10. Уникальность через столбец с повторяющимися значениями
# ============================================================
print("")
print("--- 10. Unique(bd[:, 'Имя']) ---")
r10 = Unique(bd[:, "Имя"])
print(r10)
# Ожидается: 6 строк — все имена уникальны
#   IT     Аня    Москва
#   IT     Боб    Питер
#   HR     Света  Москва
#   IT     Гена   Питер
#   HR     Даша   Москва
#   Sales  Егор   Сочи

print("")
print("=== ТЕСТ ЗАВЕРШЁН ===")
