# ============================================================
# ANOMALY — ТЕСТ БЕЗ ОШИБОК
# ============================================================
# Все рабочие сценарии:
#   1.  Базовый (iqr по умолчанию)
#   2.  iqr с параметром k=3.0
#   3.  zscore (по умолчанию N=3)
#   4.  zscore с N=2
#   5.  percentile (по умолчанию 1, 99)
#   6.  percentile с параметрами 5, 95
#   7.  by — аномалии внутри группы
#   8.  by + iqr
#   9.  only — только аномалии
#   10. approx — приблизительные процентили
#   11. only + approx
#   12. Порядок аргументов — ЛЮБОЙ
#   13. Мутация
#   14. Мало данных в группе (< 4) → __anomaly = None
#   15. Все значения одинаковые → __anomaly = 0
# ============================================================

m = ["Магазин", "Сумма";
     "М1", 1500;
     "М1", 1800;
     "М1", 2000;
     "М1", 500000;
     "М2", 300000;
     "М2", 320000;
     "М2", 310000;
     "М2", 305000]

print("=== ИСХОДНАЯ МАТРИЦА ===")
print(m)
print("")


# ------------------------------------------------------------
# 1. Базовый (iqr по умолчанию)
# ------------------------------------------------------------
print("--- 1. anomaly(m[:, 'Сумма']) ---")
r = anomaly(m[:, "Сумма"])
print(r)
print("")
# ОЖИДАНИЕ: столбец __anomaly со значениями 0/1/-1/None
# 500000 → 1 (выше верхней границы)


# ------------------------------------------------------------
# 2. iqr с k=3.0
# ------------------------------------------------------------
print("--- 2. anomaly(m[:, 'Сумма'], iqr, 3.0) ---")
r = anomaly(m[:, "Сумма"], iqr, 3.0)
print(r)
print("")
# ОЖИДАНИЕ: с k=3.0 500000 может стать 0 (мягче порог)


# ------------------------------------------------------------
# 3. zscore (по умолчанию N=3)
# ------------------------------------------------------------
print("--- 3. anomaly(m[:, 'Сумма'], zscore) ---")
r = anomaly(m[:, "Сумма"], zscore)
print(r)
print("")
# ОЖИДАНИЕ: выбросы с |z| > 3


# ------------------------------------------------------------
# 4. zscore с N=2
# ------------------------------------------------------------
print("--- 4. anomaly(m[:, 'Сумма'], zscore, 2) ---")
r = anomaly(m[:, "Сумма"], zscore, 2)
print(r)
print("")
# ОЖИДАНИЕ: с N=2 больше аномалий


# ------------------------------------------------------------
# 5. percentile (по умолчанию 1, 99)
# ------------------------------------------------------------
print("--- 5. anomaly(m[:, 'Сумма'], percentile) ---")
r = anomaly(m[:, "Сумма"], percentile)
print(r)
print("")
# ОЖИДАНИЕ: значения ниже 1% и выше 99%


# ------------------------------------------------------------
# 6. percentile с 5, 95
# ------------------------------------------------------------
print("--- 6. anomaly(m[:, 'Сумма'], percentile, 5, 95) ---")
r = anomaly(m[:, "Сумма"], percentile, 5, 95)
print(r)
print("")
# ОЖИДАНИЕ: значения ниже 5% и выше 95%


# ------------------------------------------------------------
# 7. by — аномалии внутри группы
# ------------------------------------------------------------
print("--- 7. anomaly(m[:, 'Сумма'], by m[:, 'Магазин']) ---")
r = anomaly(m[:, "Сумма"], by m[:, "Магазин"])
print(r)
print("")
# ОЖИДАНИЕ:
# М1: 1500, 1800, 2000 — норма; 500000 — аномалия (1)
# М2: 300000, 320000, 310000, 305000 — все норма (0)


# ------------------------------------------------------------
# 8. by + iqr
# ------------------------------------------------------------
print("--- 8. anomaly(..., by ..., iqr) ---")
r = anomaly(m[:, "Сумма"], by m[:, "Магазин"], iqr)
print(r)
print("")


# ------------------------------------------------------------
# 9. only — только аномалии
# ------------------------------------------------------------
print("--- 9. anomaly(m[:, 'Сумма'], only) ---")
r = anomaly(m[:, "Сумма"], only)
print(r)
print("")
# ОЖИДАНИЕ: только строки с __anomaly != 0 и не None


# ------------------------------------------------------------
# 10. approx — приблизительные процентили
# ------------------------------------------------------------
print("--- 10. anomaly(m[:, 'Сумма'], percentile, approx) ---")
r = anomaly(m[:, "Сумма"], percentile, approx)
print(r)
print("")


# ------------------------------------------------------------
# 11. only + approx
# ------------------------------------------------------------
print("--- 11. anomaly(m[:, 'Сумма'], only, approx) ---")
r = anomaly(m[:, "Сумма"], only, approx)
print(r)
print("")


# ------------------------------------------------------------
# 12. Порядок аргументов — ЛЮБОЙ
# ------------------------------------------------------------
print("--- 12. anomaly(..., only, by ..., zscore, 2, approx) ---")
r = anomaly(m[:, "Сумма"],
            only,
            by m[:, "Магазин"],
            zscore,
            2,
            approx)
print(r)
print("")
# ОЖИДАНИЕ: те же результаты, что у теста 11, но через zscore


# ------------------------------------------------------------
# 13. Мутация
# ------------------------------------------------------------
print("--- 13. мутация m2 = anomaly(...) ---")
m2 = ["Магазин", "Сумма";
      "М1", 100;
      "М1", 110;
      "М1", 120;
      "М1", 9000]

m2 = anomaly(m2[:, "Сумма"])
print(m2)
print("")


# ------------------------------------------------------------
# 14. Мало данных в группе (< 4) → __anomaly = None
# ------------------------------------------------------------
print("--- 14. anomaly с малой группой ---")
m3 = ["Магазин", "Сумма";
      "М1", 100;
      "М1", 110;
      "М1", 120;
      "М1", 130;
      "М2", 500;
      "М2", 600;
      "М2", 700]

r = anomaly(m3[:, "Сумма"], by m3[:, "Магазин"])
print(r)
print("")
# ОЖИДАНИЕ:
# М1: 4 значения → считаем аномалии
# М2: 3 значения → __anomaly = None


# ------------------------------------------------------------
# 15. Все значения одинаковые → __anomaly = 0
# ------------------------------------------------------------
print("--- 15. anomaly одинаковых значений ---")
m4 = ["Магазин", "Сумма";
      "М1", 100;
      "М1", 100;
      "М1", 100;
      "М1", 100;
      "М1", 100]

r = anomaly(m4[:, "Сумма"])
print(r)
print("")
# ОЖИДАНИЕ: все __anomaly = 0 (IQR = 0, границы None → 0)


print("=== ВСЕ ТЕСТЫ ANOMALY (OK) ЗАВЕРШЕНЫ ===")
