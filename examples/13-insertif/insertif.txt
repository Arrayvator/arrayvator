# ============================================================
# ТЕСТЫ INSERTIF
# ============================================================

# ------------------------------------------------------------
# ТЕСТ 1: Базовый — вставить после каждой строки с отделом IT
# ------------------------------------------------------------
println("=== ТЕСТ 1: insertif(m[:, 'Отдел'] == 'IT', after) ===")

m = ["Имя", "Отдел", "Зарплата";
     "Аня", "IT",    85000;
     "Боб", "HR",    65000;
     "Света","IT",   95000;
     "Гена","Sales", 70000;
     "Даша","IT",    88000]

println("Исходная:")
println(m)
println("")

r = insertif(m[:, "Отдел"] == "IT", after)
println("После вставки:")
println(r)
println("")

# ОЖИДАНИЕ:
# Имя    Отдел  Зарплата
# Аня    IT     85000
# None   None   None      ← вставлено
# Боб    HR     65000
# Света  IT     95000
# None   None   None      ← вставлено
# Гена   Sales  70000
# Даша   IT     88000
# None   None   None      ← вставлено


# ------------------------------------------------------------
# ТЕСТ 2: before — вставить ПЕРЕД каждой строкой с IT
# ------------------------------------------------------------
println("=== ТЕСТ 2: insertif(m[:, 'Отдел'] == 'IT', before) ===")

r = insertif(m[:, "Отдел"] == "IT", before)
println(r)
println("")

# ОЖИДАНИЕ:
# Имя    Отдел  Зарплата
# None   None   None      ← вставлено
# Аня    IT     85000
# Боб    HR     65000
# None   None   None      ← вставлено
# Света  IT     95000
# Гена   Sales  70000
# None   None   None      ← вставлено
# Даша   IT     88000


# ------------------------------------------------------------
# ТЕСТ 3: Числовое условие
# ------------------------------------------------------------
println("=== ТЕСТ 3: insertif(m[:, 'Зарплата'] > 80000, after) ===")

r = insertif(m[:, "Зарплата"] > 80000, after)
println(r)
println("")

# ОЖИДАНИЕ:
# Аня (85000) → да
# Боб (65000) → нет
# Света (95000) → да
# Гена (70000) → нет
# Даша (88000) → да


# ------------------------------------------------------------
# ТЕСТ 4: Составное условие (AND)
# ------------------------------------------------------------
println("=== ТЕСТ 4: insertif(IT AND > 90000, after) ===")

r = insertif(m[:, "Отдел"] == "IT" and m[:, "Зарплата"] > 90000, after)
println(r)
println("")

# ОЖИДАНИЕ: только Света (IT, 95000)


# ------------------------------------------------------------
# ТЕСТ 5: Составное условие (OR)
# ------------------------------------------------------------
println("=== ТЕСТ 5: insertif(HR OR Sales, after) ===")

r = insertif(m[:, "Отдел"] == "HR" or m[:, "Отдел"] == "Sales", after)
println(r)
println("")

# ОЖИДАНИЕ: Боб (HR), Гена (Sales)


# ------------------------------------------------------------
# ТЕСТ 6: NOT
# ------------------------------------------------------------
println("=== ТЕСТ 6: insertif(not IT, after) ===")

r = insertif(not (m[:, "Отдел"] == "IT"), after)
println(r)
println("")

# ОЖИДАНИЕ: Боб (HR), Гена (Sales)


# ------------------------------------------------------------
# ТЕСТ 7: Нет совпадений — вернуть как есть
# ------------------------------------------------------------
println("=== ТЕСТ 7: insertif(Отдел == 'XYZ', after) — нет совпадений ===")

r = insertif(m[:, "Отдел"] == "XYZ", after)
println(r)
println("")

# ОЖИДАНИЕ: матрица без изменений


# ------------------------------------------------------------
# ТЕСТ 8: ВЕКТОР
# ------------------------------------------------------------
println("=== ТЕСТ 8: insertif(v > 20, after) — вектор ===")

v = [10, 30, 15, 40, 20, 50]
println("Исходный вектор:")
print(v)
println("")

r = insertif(v > 20, after)
print(r)
println("")

# ОЖИДАНИЕ:
# 10, 30, None, 15, 40, None, 20, 50, None
# Вставляем после 30, 40, 50


# ------------------------------------------------------------
# ТЕСТ 9: Вектор + before
# ------------------------------------------------------------
println("=== ТЕСТ 9: insertif(v > 20, before) ===")

r = insertif(v > 20, before)
print(r)
println("")

# ОЖИДАНИЕ:
# 10, None, 30, 15, None, 40, 20, None, 50


# ------------------------------------------------------------
# ТЕСТ 10: Все строки совпадают
# ------------------------------------------------------------
println("=== ТЕСТ 10: insertif(все True, after) ===")

m2 = ["A"; 1; 2; 3]
r = insertif(m2[:, "A"] > 0, after)
println(r)
println("")

# ОЖИДАНИЕ: после каждой строки вставлена пустая
# A
# 1
# None
# 2
# None
# 3
# None


# ------------------------------------------------------------
# ТЕСТ 11: Мутация
# ------------------------------------------------------------
println("=== ТЕСТ 11: мутация m = insertif(...) ===")

m3 = ["Имя", "Отдел";
      "Аня", "IT";
      "Боб", "HR"]

m3 = insertif(m3[:, "Отдел"] == "IT", after)
println(m3)
println("")

# ОЖИДАНИЕ:
# Имя   Отдел
# Аня   IT
# None  None    ← вставлено
# Боб   HR
