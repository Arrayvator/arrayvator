m4 = matrix(5, 1)
m4[:, 1] = range(1:10, step 2)
print("range(1:10, step 2) =", m4[:, 1])

# ТЕСТ 7: joinarray
# ============================================================
print("=== ТЕСТ 7: joinarray ===")
a = ["A", "B"; 1, 2]
b = ["C", "D"; 3, 4]
print("a =", a)
print("b =", b)

rv = joinarray(a, b, vertical)
print("joinarray(a, b, vertical) =")
print(rv)

rh = joinarray(a, b, horizontal)
print("joinarray(a, b, horizontal) =")
