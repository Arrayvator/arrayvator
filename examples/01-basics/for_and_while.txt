# 6. Обход матрицы по строкам
# ============================================================

# ============================================================
# 7. Фильтрация через for — только те, кому больше 30
# ============================================================

}

# 11. while + break: первое i*i > 10
# ============================================================

# ============================================================
# 14. Вложенные циклы: сумма всех чисел матрицы
# ============================================================
println("=== 14. Сумма всех чисел матрицы ===")
m = ["A", "B", "C";
     1,   2,   3;
     4,   5,   6;
     7,   8,   9]

s = 0
for i = 2 to lenrow(m) do {
    for j = 1 to lencol(m) do {
        x = m[i, j]
        if is_number(x) then {
            s = s + x
        }
    }
}
print("Итого:", s)
println("")

# ============================================================
# 16. while + матрица: скидка 10%
# ============================================================
println("=== 16. Скидка 10% через while ===")
m = ["Товар",  "Цена";
     "Яблоки", 100;
     "Бананы", 80;
     "Груши",  150]

i = 2
while i <= lenrow(m) {
    x = m[i, 2]
    m[i, 2] = round(x * 0.9, 2)
    i = i + 1
}
print(m)


# ============================================================
# 17. for + applyif (сравнение двух способов)
# ============================================================
println("=== 17. for + условия: Джун/Сеньор ===")
m = ["Имя",   "Возраст";
     "Аня",   25;
     "Боб",   32;
     "Света", 28]

for i = 2 to lenrow(m) do {
    if m[i, 2] > 30 then {
        m[i, 3] = "Сеньор"
    } else {
        m[i, 3] = "Джун"
    }
}
print(m)

# 20. for + ветвление: чётное/нечётное

println("=== 20. Чётное или нечётное ===")
m = ["Число", "Тип";
     1,       "";
     2,       "";
     3,       "";
     4,       "";
     5,       "";
     6,       ""]

for i = 2 to lenrow(m) do {

# 22. while + break: ищем первое None
# ============================================================
println("=== 22. while + break: первое None ===")
m = ["Имя",   "Возраст";
     "Аня",   25;
     "Боб",   None;
     "Света", 28]

i = 2
while i <= lenrow(m) {
    if isnone(m[i, 2]) then {
        print("Первое None в строке", i, ":", m[i, 1])
        break
    }
    i = i + 1
}
println("")

# 27. for: умножение матрицы на число
# ============================================================
println("=== 27. Умножить все числа на 2 ===")
m = ["A", "B", "C";
     1,   2,   3;
     4,   5,   6]

for i = 2 to lenrow(m) do {
    for j = 1 to lencol(m) do {

            m[i, j] = x * 2
        }
    }
}

# 28. while + накопление минимума
# ============================================================
println("=== 28. Минимум через while ===")
v = [42, 17, 93, 5, 68, 31]
i = 1
x = v[1]
while i <= len(v) {
    if v[i] < x then {
        x = v[i]
    }
    i = i + 1
}
print("Минимум:", x)      # 5
println("")
