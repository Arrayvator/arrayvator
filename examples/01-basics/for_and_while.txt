# Циклы

# for: один оператор — без скобок
for i = 1 to 5 do print(i)
print()
# После:
# 1 2 3 4 5


# for: обратный отсчёт
for i = 5 to 1 do print(i)
print()
# После:
# 5 4 3 2 1


# for: чётные (шаг 2)
for i = 2 to 10 step 2 do print(i)
print()
# После:
# 2 4 6 8 10


# for: шаг 3
for i = 1 to 10 step 3 do print(i)
print()
# После:
# 1 4 7 10


# for: убывающий с шагом -2
for i = 10 to 1 step -2 do print(i)
print()
# После:
# 10 8 6 4 2


# for: несколько операторов — с блоком

# Обход матрицы по строкам
m = ["Имя",   "Возраст";
     "Аня",   25;
     "Боб",   32;
     "Света", 28;
     "Гена",  41]
for i = 2 to lenrow(m) do {
    print("Строка", i, ":", m[i, 1], m[i, 2])
}
# После:
# Строка 2 : Аня 25
# Строка 3 : Боб 32
# Строка 4 : Света 28
# Строка 5 : Гена 41

# while + break

    if i * i > 10 then {

# Вложенные циклы: сумма всех чисел матрицы
m = ["A", "B", "C";
     1,   2,   3;
     4,   5,   6;
     7,   8,   9]
s = 0
for i = 2 to lenrow(m) do {
    for j = 1 to lencol(m) do {
        x = m[i, j]
        if is_number(x) then s = s + x
    }
}
println(s)

# 45

# while + матрица: скидка 10%
m = ["Товар",  "Цена";
     "Яблоки", 100;
     "Бананы", 80;
     "Груши",  150]
i = 2
while i <= lenrow(m) do {
    x = m[i, 2]
    m[i, 2] = round(x * 0.9, 2)
    i = i + 1
}
print(m)
# После:
# Товар   Цена
# Яблоки  90.0
# Бананы  72.0
# Груши   135.0


# for + условия: Джун/Сеньор

     "Аня",   25;
     "Боб",   32;
     "Света", 28]
for i = 2 to lenrow(m) do {
    if m[i, 2] > 30 then {
        m[i, 3] = "Сеньор"
    } else {
        m[i, 3] = "Джун"
    }
}
println(m)

# Имя    Возраст  None
# Аня    25       Джун
# Боб    32       Сеньор
# Света  28       Джун

# for + ветвление: чётное/нечётное
m = ["Число", "Тип";
     1,       "";
     2,       "";
     3,       "";
     4,       "";
     5,       "";
     6,       ""]
for i = 2 to lenrow(m) do {
    n = m[i, 1]
    if n % 2 == 0 then {
        m[i, 2] = "чётное"
    } else {
        m[i, 2] = "нечётное"
    }
}
print(m)
# После:
# Число  Тип
# 1      нечётное
# 2      чётное
# 3      нечётное
# 4      чётное
# 5      нечётное
# 6      чётное

# while + break: ищем первое None
m = ["Имя",   "Возраст";
     "Аня",   25;
     "Боб",   None;
     "Света", 28]
i = 2
while i <= lenrow(m) do {
    if isnone(m[i, 2]) then {
        println("Первое None в строке", i, ":", m[i, 1])
        break
    }
    i = i + 1
}
# После:
# Первое None в строке 3 : Боб

# for: умножение матрицы на число
m = ["A", "B", "C";
     1,   2,   3;
     4,   5,   6]
for i = 2 to lenrow(m) do {
    for j = 1 to lencol(m) do {

    }
