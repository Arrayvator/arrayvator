# ============================================================
# ТЕСТ 1: type() — определение типа значения
# ============================================================
print("=== ТЕСТ 1: type() ===")
print("type(42)      =", type(42))
print("type(3.14)    =", type(3.14))
print("type(\"abc\")   =", type("abc"))
print("type(true)    =", type(true))
print("type(false)   =", type(false))
print("type(None)    =", type(None))

v = [1, 2, 3]
print("type(v)       =", type(v))

m = ["A", "B"; 1, 2]
print("type(m)       =", type(m))
print("")

# ТЕСТ 2: is_number() — проверка на число
# ============================================================
print("=== ТЕСТ 2: is_number() ===")
print("is_number(42)     =", is_number(42))
print("is_number(3.14)   =", is_number(3.14))
print("is_number(\"42\")   =", is_number("42"))
print("is_number(true)   =", is_number(true))
print("is_number(None)   =", is_number(None))

v2 = [10, "abc", 3.14, true, None]
r2 = is_number(v2)
print("is_number([10, \"abc\", 3.14, true, None])")
print("  →", r2)
print("")
