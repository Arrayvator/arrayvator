# Изменение таблицы и ячеек

m = ["Имя", "Зарплата";
     "Аня", 50000;
     "Боб", 60000;
     "Света", 55000]
print(m)
print()

# Если записать название стобца, который не сущесетвует
# то столбец с таким названием автоматичеси добавится
m[:,"Налог"]=0 # В новый стобец записали нули
print(m)
print()

# через end+1, т.е. последний столбец + 1
# через этот способ у столбца нет имени
m[:,end+1]="Добавка"
m[1,end]="Последний столбец" # Добавили имя столбца "Последний столбец"
print(m)
print()

m[:,end+1]=range(1:lenrow(m))
m[1,end]="Номер попорядку"
print(m)
print()

# Если имя строки не найдено, то автоматически добавляется строка 
# с таким именем которые вы ввели
m["Андрей",:]="Добавка в строку" # Во всю строку приравняли единицу
print(m)
print()

# Также строки и столбцы можно добавлять номером за пределами таблицы или 
# через end+1, т.е. последняя строка/столбец + 1
m[end+1,:]="Последняя строка" 
print(m)
print()
