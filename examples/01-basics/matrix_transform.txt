# Изменение таблицы и ячеек

# Исходная таблица
m = ["Имя", "Зарплата";
     "Аня", 50000;
     "Боб", 60000;
     "Света", 55000]
println(m)
# После:
# Имя    Зарплата
# Аня    50000
# Боб    60000
# Света  55000

println()
# Добавить новый столбец "Налог" = 0
m = ["Имя", "Зарплата";
     "Аня", 50000;
     "Боб", 60000;
     "Света", 55000]
m[:, "Налог"] = 0
println(m)
# После:
# Имя    Зарплата  Налог
# Аня    50000     0
# Боб    60000     0
# Света  55000     0

println()
# Добавить новый столбец через end+1 (без имени)
m = ["Имя", "Зарплата";
     "Аня", 50000;
     "Боб", 60000;
     "Света", 55000]
m[:, end+1] = "Добавка"
m[1, end] = "Последний столбец"
println(m)
# После:
# Имя    Зарплата  Последний столбец
# Аня    50000     Добавка
# Боб    60000     Добавка
# Света  55000     Добавка

println()
# Добавить столбец с range
m = ["Имя", "Зарплата";
     "Аня", 50000;
     "Боб", 60000;
     "Света", 55000]
m[:, end+1] = range(1:lenrow(m))
m[1, end] = "Номер попорядку"
println(m)
# После:
# Имя    Зарплата  Номер попорядку
# Аня    50000     1
# Боб    60000     2
# Света  55000     3

println()
# Добавить строку через номер за пределами (end+1)
m = ["Имя", "Зарплата";
     "Аня", 50000;
     "Боб", 60000;
     "Света", 55000]
m[end+1, :] = "Последняя строка"
println(m)
# После:
# Имя    Зарплата
# Аня    50000
# Боб    60000
# Света  55000
# None   Последняя строка

println()
# Добавить строку через явный номер 5
m = ["Имя", "Зарплата";
     "Аня", 50000;
     "Боб", 60000;
     "Света", 55000]
m[5, :] = "Новая строка"
println(m)
# После:
# Имя    Зарплата
# Аня    50000
# Боб    60000
# Света  55000
# None   Новая строка

println()
