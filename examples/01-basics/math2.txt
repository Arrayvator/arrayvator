println("============================================================")
println("БЛОК 1. МАТРИЦА ↔ СКАЛЯР  (+ - * / %)")
println("============================================================")

m = [
    "A", "B";
    10,  20;
    30,  40
]

println("Исходная матрица m:")
print(m)
println()

# 1.1 Сложение
println("1.1  m + 5  (ожидаем 15 25 / 35 45):")
print(m + 5)
println()

# 1.2 Вычитание
println("1.2  m - 5  (ожидаем 5 15 / 25 35):")
print(m - 5)
println()

# 1.3 Умножение
println("1.3  m * 2  (ожидаем 20 40 / 60 80):")
print(m * 2)
println()

# 1.4 Деление
println("1.4  m / 2  (ожидаем 5.0 10.0 / 15.0 20.0):")
print(m / 2)
println()

# 1.5 Остаток
println("1.5  m % 7  (ожидаем 3 6 / 2 5):")
print(m % 7)
println()

# 1.6 Скаляр слева — вычитание
println("1.6  100 - m  (ожидаем 90 80 / 70 60):")
print(100 - m)
println()

# 1.7 Скаляр слева — деление
println("1.7  100 / m  (ожидаем 10.0 5.0 / 3.3333 2.5):")
print(100 / m)
println()

# 1.8 Скаляр слева — остаток
println("1.8  100 % m  (ожидаем 0 0 / 10 20):")
print(100 % m)
println()

println("БЛОК 3. ЗАПИСЬ РЕЗУЛЬТАТА В НОВЫЙ СТОЛБЕЦ")
println("============================================================")

m = [
    "A", "B";
    10,  20;
    30,  40
]

println("Исходная матрица m:")
print(m)
println()

# 3.1 Сложение двух столбцов
m[:, "Сумма"] = m[:, "A"] + m[:, "B"]
println("3.1  m[:,Сумма] = m[:,A] + m[:,B]  (ожидаем 30, 70)")
print(m)
println()

# 3.2 Вычитание двух столбцов
m[:, "Разность"] = m[:, "A"] - m[:, "B"]
println("3.2  m[:,Разность] = m[:,A] - m[:,B]  (ожидаем -10, -10)")
print(m)
println()

# 3.3 Умножение двух столбцов
m[:, "Произведение"] = m[:, "A"] * m[:, "B"]
println("3.3  m[:,Произведение] = m[:,A] * m[:,B]  (ожидаем 200, 1200)")
print(m)
println()

# 3.4 Деление двух столбцов
m[:, "Частное"] = m[:, "B"] / m[:, "A"]
println("3.4  m[:,Частное] = m[:,B] / m[:,A]  (ожидаем 2.0, 1.3333)")
print(m)
println()

# 3.5 Остаток
m[:, "Остаток"] = m[:, "B"] % m[:, "A"]
println("3.5  m[:,Остаток] = m[:,B] % m[:,A]  (ожидаем 0, 10)")
print(m)
println()

println("============================================================")
println("БЛОК 5. КОМБИНИРОВАННЫЕ ВЫРАЖЕНИЯ (без **)")
println("============================================================")

n = [
    "X", "Y";
    2,   3;
    4,   5;
    6,   7
]

println("Исходная матрица n:")
print(n)
println()

# 5.1 (X + Y) * 2
n[:, "A"] = (n[:, "X"] + n[:, "Y"]) * 2
println("5.1  (X + Y) * 2     =  10, 18, 26")
print(n)
println()

# 5.2 X * Y - X
n[:, "B"] = n[:, "X"] * n[:, "Y"] - n[:, "X"]
println("5.2  X * Y - X       =  4, 16, 36")
print(n)
println()

# 5.3 Y / X + 1
n[:, "C"] = n[:, "Y"] / n[:, "X"] + 1
println("5.3  Y / X + 1       =  2.5, 2.25, 2.1667")
print(n)
println()

# 5.4 (X + Y) / (Y - X)
n[:, "D"] = (n[:, "X"] + n[:, "Y"]) / (n[:, "Y"] - n[:, "X"])
println("5.4  (X + Y)/(Y - X) =  5.0, 9.0, 13.0")
print(n)
println()

# 5.5 X % Y
n[:, "E"] = n[:, "X"] % n[:, "Y"]
println("5.5  X % Y           =  2, 4, 6")
print(n)
println()

# 5.6 X * X + Y * Y  (замена X**2 + Y**2)
n[:, "F"] = n[:, "X"] * n[:, "X"] + n[:, "Y"] * n[:, "Y"]
println("5.6  X*X + Y*Y       =  13, 41, 85")
print(n)
println()
