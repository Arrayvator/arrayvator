println("============================================================")
println("БЛОК 1. СКАЛЯР ↔ СКАЛЯР")
println("============================================================")

println("1.1  Сложение:      2 + 3       = ", 2 + 3)            # 5
println("1.2  Вычитание:     10 - 4      = ", 10 - 4)           # 6
println("1.3  Умножение:     6 * 7       = ", 6 * 7)            # 42
println("1.4  Деление:       15 / 4      = ", 15 / 4)           # 3.75
println("1.5  Остаток:       10 % 3      = ", 10 % 3)           # 1
println("1.6  Деление на 0:  10 / 0      = ", 10 / 0)           # inf
println("1.7  Остаток на 0:  10 % 0      = ", 10 % 0)           # inf
println()


# ============================================================
println("============================================================")
println("БЛОК 2. ВЕКТОР ↔ СКАЛЯР")
println("============================================================")

v = [10, 20, 30, 40]
s = 5

println("Дан вектор v  = ", v)
println("Дан скаляр s  = ", s)
println()

println("2.1  v + s = ", v + s)            # [15, 25, 35, 45]
println("2.2  v - s = ", v - s)            # [5, 15, 25, 35]
println("2.3  v * s = ", v * s)            # [50, 100, 150, 200]
println("2.4  v / s = ", v / s)            # [2, 4, 6, 8]
println("2.5  v % 3 = ", v % 3)            # [1, 2, 0, 1]
println()

# Скаляр слева
println("2.6  s + v = ", s + v)            # [15, 25, 35, 45]
println("2.7  s - v = ", s - v)            # [-5, -15, -25, -35]
println("2.8  s * v = ", s * v)            # [50, 100, 150, 200]
println("2.9  s / v = ", s / v)            # [0.5, 0.25, ...]
println()

println("============================================================")
println("БЛОК 3. ВЕКТОР ↔ ВЕКТОР")

a = [1, 2, 3, 4]

# ============================================================
println("============================================================")
println("БЛОК 4. СТОЛБЦЫ МАТРИЦЫ ↔ СКАЛЯР")
println("============================================================")

m = [
    "Товар",  "Цена",  "Кол-во";
    "A",      100,     3;
    "B",      250,     2;
    "C",      400,     5;
    "D",      50,      10
]

println("Исходная матрица m:")
print(m)
println()

println("4.1  m[:,Цена] + 100     = ", m[:, "Цена"] + 100)
println("4.2  m[:,Цена] - 50      = ", m[:, "Цена"] - 50)
println("4.3  m[:,Цена] * 2       = ", m[:, "Цена"] * 2)
println("4.4  m[:,Цена] / 10      = ", m[:, "Цена"] / 10)
println("4.5  m[:,Цена] % 100     = ", m[:, "Цена"] % 100)
println()


# ============================================================
println("============================================================")
println("БЛОК 5. СТОЛБЕЦ ↔ СТОЛБЕЦ (АРИФМЕТИКА МЕЖДУ СТОЛБЦАМИ)")
println("============================================================")

# 5.1 СУММА
m[:, "Сумма"] = m[:, "Цена"] + m[:, "Кол-во"]
println("5.1  m[:,Цена] + m[:,Кол-во] = ", m[:, "Сумма"])

# 5.2 РАЗНОСТЬ
m[:, "Разность"] = m[:, "Цена"] - m[:, "Кол-во"]
println("5.2  m[:,Цена] - m[:,Кол-во] = ", m[:, "Разность"])

# 5.3 ПРОИЗВЕДЕНИЕ
m[:, "Итого"] = m[:, "Цена"] * m[:, "Кол-во"]
println("5.3  m[:,Цена] * m[:,Кол-во] = ", m[:, "Итого"])

# 5.4 ДЕЛЕНИЕ
m[:, "Отношение"] = m[:, "Цена"] / m[:, "Кол-во"]
println("5.4  m[:,Цена] / m[:,Кол-во] = ", m[:, "Отношение"])

# 5.5 ОСТАТОК
m[:, "Остаток"] = m[:, "Цена"] % m[:, "Кол-во"]
println("5.5  m[:,Цена] % m[:,Кол-во] = ", m[:, "Остаток"])

println()
println("Матрица m после добавления столбцов:")
print(m)
println()


# ============================================================
println("============================================================")
println("БЛОК 6. КОМБИНИРОВАННЫЕ ВЫРАЖЕНИЯ")
println("============================================================")

n = [
    "X",  "Y";
    2,    3;
    4,    5;
    6,    7
]

println("Исходная матрица n:")
print(n)
println()

# 6.1 (X + Y) * 2
n[:, "A"] = (n[:, "X"] + n[:, "Y"]) * 2
println("6.1  (X + Y) * 2     = ", n[:, "A"])          # [10, 18, 26]

# 6.2 X * Y - X
n[:, "B"] = n[:, "X"] * n[:, "Y"] - n[:, "X"]
println("6.2  X * Y - X       = ", n[:, "B"])          # [4, 16, 36]

# 6.3 Y / X + 1
n[:, "C"] = n[:, "Y"] / n[:, "X"] + 1
println("6.3  Y / X + 1       = ", n[:, "C"])          # [2.5, 2.25, 2.166...]

# 6.4 (X + Y) / (Y - X)
n[:, "D"] = (n[:, "X"] + n[:, "Y"]) / (n[:, "Y"] - n[:, "X"])
println("6.4  (X + Y)/(Y - X) = ", n[:, "D"])          # [5, 9, 13]

# 6.5 X % Y
n[:, "E"] = n[:, "X"] % n[:, "Y"]
println("6.5  X % Y            = ", n[:, "E"])         # [2, 4, 6]

println()
println("Матрица n после добавления столбцов:")
print(n)
println()

println("БЛОК 9. ДЕМО-ПРИМЕР: НАДБАВКА ЗА СТАЖ")
println("============================================================")

e = [
    "Сотрудник", "Зарплата", "Стаж";
    "Аня",       80000,      5;
    "Боб",       80000,      3;
    "Света",     80000,      10;
    "Гена",      80000,      8
]

println("Исходная матрица e:")
print(e)
println()

e[:, "Надбавка"] = e[:, "Зарплата"] * (e[:, "Стаж"] / 100)
e[:, "Итого"]    = e[:, "Зарплата"] + e[:, "Надбавка"]
e[:, "Проверка"] = e[:, "Итого"] - e[:, "Зарплата"] - e[:, "Надбавка"]   # должно быть 0

println("Матрица после расчётов:")
print(e)
println()

println("Проверка (должны быть нули): ", e[:, "Проверка"])
println()
