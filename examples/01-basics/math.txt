# round: вектор
v = [3.7, -2.5, 1.234, 10.0]
r = round(v)
print(r)
println()
# После:
# 4 -2 1 10


# round: вектор с точностью
v = [3.7, -2.5, 1.234, 10.0]
r = round(v, 1)
print(r)
println()
# После:
# 3.7 -2.5 1.2 10.0

# round: матрица с точностью
m = ["A", "B";
     1.234, 5.678;
     -2.5, 3.7]
r = round(m, 1)
println(r)
# После:
# A     B
# 1.2   5.7
# -2.5  3.7


# int: матрица
m = ["A", "B";
     1.234, 5.678;
     -2.5, 3.7]
r = int(m)
println(r)
# После:
# A   B
# 1   5
# -2  3


# Срез столбца через addcolumn
m2 = ["X", "Y";
      1.11, 2.22;
      3.33, 4.44]
m3 = addcolumn(m2, "Окр", round(m2[:, "X"], 1))
println(m3)
# После:
# X     Y     Окр
# 1.11  2.22  1.1
# 3.33  4.44  3.3
