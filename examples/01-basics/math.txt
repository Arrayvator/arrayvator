print("=== 1.1. Скаляры ===")
print("round(3.14159)      =", round(3.14159))
print("round(3.14159, 2)   =", round(3.14159, 2))
print("round(2.5)          =", round(2.5))
print("round(3.5)          =", round(3.5))
print("round(-3.7)         =", round(-3.7))
print("")

print("int(3.7)            =", int(3.7))
print("int(-3.7)           =", int(-3.7))
print("int(42)             =", int(42))
print("")

print("frac(3.7)           =", frac(3.7))
print("frac(-3.7)          =", frac(-3.7))
print("frac(3.14159, 2)    =", frac(3.14159, 2))
print("")

print("frac_digits(3.456)  =", frac_digits(3.456))
print("frac_digits(3.14)   =", frac_digits(3.14))
print("frac_digits(3.0)    =", frac_digits(3.0))
print("")


print("=== 1.2. Вектор ===")
v = [3.7, -2.5, 1.234, 10.0]
print("v =", v)
print("round(v)       =", round(v))
print("round(v, 1)    =", round(v, 1))
print("int(v)         =", int(v))
print("frac(v)        =", frac(v))
print("frac_digits(v) =", frac_digits(v))
print("")


print("=== 1.3. Матрица ===")
m = ["A", "B";
     1.234, 5.678;
     -2.5, 3.7]
print("Исходная:")
print(m)
print("")

print("round(m, 1):")
print(round(m, 1))
print("")

print("int(m):")
print(int(m))
print("")


print("=== 1.4. Срез столбца через addcolumn ===")
m2 = ["X", "Y";
      1.11, 2.22;
      3.33, 4.44]
m3 = addcolumn(m2, "Окр", round(m2[:, "X"], 1))
print(m3)
print("")
