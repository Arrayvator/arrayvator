# Строки

# split: без разделителя — по символам
r = split("Hello")
print(r)
println()
# После:
# H e l l o


# split: с разделителем
r = split("a-b-c", "-")
print(r)
println()
# После:
# a b c


# split: skip пустых
r = split("a,,b,,c", ",", skip)
print(r)
println()
# После:
# a b c


# split: с разделителем ;
r = split("a;b;c", ";")
print(r)
println()
# После:
# a b c


# joinvector: со строками
r = joinvector(["a", "b", "c"], "-")
print(r)
println()
# После:
# a-b-c


# joinvector: с числами
r = joinvector([10, 20, 30], ", ")
print(r)
println()
# После:
# 10, 20, 30


# joinvector: без разделителя — склейка
r = joinvector(["H", "i"])
print(r)
println()
# После:
# Hi


# joinvector: вектор-переменная
v = split("one two three", " ")
r = joinvector(v, "_")
print(r)
println()
# После:
# one_two_three


# replacetext: простая замена в строке
r = replacetext("Hello", "l", "p")
print(r)
println()
# После:
# Heppo


# replacetext: в столбце, присваивание в столбец
m = ["Имя", "Отдел";
     "Аня",  "IT";
     "Боб",  "HR";
     "Света", "IT";
     "Гена", "Sales"]
m[:, "Отдел"] = replacetext(m[:, "Отдел"], "IT", "--")
println(m)
# После:
# Имя    Отдел
# Аня    --
# Боб    HR
# Света  --
# Гена   Sales


# replacetext: ignore — без учёта регистра
m = ["Имя", "Отдел";
     "Аня",  "IT";
     "Боб",  "HR";
     "Света", "IT";
     "Гена", "Sales"]
r = replacetext(m[:, "Отдел"], "it", "--", ignore)
print(r)
println()
# После:
# -- HR -- Sales


# trim: с обеих сторон
r = trim("  hello  ")
print(r)
println()
# После:
# hello


# trim: с символами
r = trim("...hello...", ".")
print(r)
println()
# После:
# hello


# trimleft
r = trimleft("  hello  ")
print(r)
println()
# После:
# hello   


# trimright
r = trimright("  hello  ")
print(r)
println()
# После:
# hello


# trimleft: с символами
r = trimleft("00042", "0")
print(r)
println()
# После:
# 42


# trimright: с символами
r = trimright("42.00.00", "0")
print(r)
println()
# После:
# 42.00.


# trim: None остаётся None
r = trim(None)
print(r)
println()
# После:
# None


# trim: числа не трогаются
r = trim(42)
print(r)
println()
# После:
# 42


# trim: вектор

      "  Аня  ";

      "Вера"]

# clean: в столбце
m5 = ["Код";
      "+7 (999) 123-45-67";
      "8-800-555-35-35";
      "(495) 123-45-67"]
r = clean(m5[:, "Код"], "digits")
print(r)
println()
# После:
# 79991234567 88005553535 4951234567
