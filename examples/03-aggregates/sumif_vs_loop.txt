# ============================================================
# ДЕМО: sumif  vs  цикл for  vs  обратный цикл + delete
# ============================================================
# Задача: посчитать СУММУ зарплат сотрудников отдела "IT"
# ============================================================

# ------------------------------------------------------------
# 1. ИСХОДНЫЕ ДАННЫЕ
# ------------------------------------------------------------
m = ["Отдел", "Сотрудник", "Зарплата";
     "IT",    "Аня",       85000;
     "HR",    "Боб",       65000;
     "IT",    "Света",     95000;
     "Sales", "Гоша",      80000;
     "IT",    "Даша",      70000;
     "HR",    "Егор",      72000]

print("=== ИСХОДНАЯ ТАБЛИЦА ===")
print(m)


# ------------------------------------------------------------
# 2. СПОСОБ 1 — sumif (декларативно, в одну строку)
# ------------------------------------------------------------
print()
print("=== СПОСОБ 1: sumif ===")

a = sumif(m[:, "Отдел"] == "IT", m[:, "Зарплата"])
print("Сумма зарплат IT:", a)


# ------------------------------------------------------------
# 3. СПОСОБ 2 — прямой цикл for (считаем вручную)
# ------------------------------------------------------------
print()
print("=== СПОСОБ 2: цикл for ===")

s = 0
for i = 2 to lenrow(m) do
    if m[i, "Отдел"] == "IT" then
        s = s + m[i, "Зарплата"]



print("Сумма зарплат IT:", s)


# ------------------------------------------------------------
# 4. СПОСОБ 3 — обратный цикл + delete (сумма через удаление)
# ------------------------------------------------------------
print()
print("=== СПОСОБ 3: обратный цикл + delete ===")

# 4.1. Копируем матрицу (присваивание делает копию)
b = m

# 4.2. Идём С КОНЦА ВВЕРХ и удаляем всё, что НЕ "IT"
for i = lenrow(b) to 2 do
{
    if b[i, "Отдел"] != "IT" then
    {
        b = delete(b[i, :])
    }
}

# 4.3. Остались только IT — суммируем столбец "Зарплата"
#      Столбец "Зарплата" — это b[:, "Зарплата"] (без заголовка)
k = sum(b[:, "Зарплата"])
print("Сумма зарплат IT:", k)


# ------------------------------------------------------------
# 5. ПРОВЕРКА: ВСЕ ТРИ РЕЗУЛЬТАТА ДОЛЖНЫ СОВПАСТЬ
# ------------------------------------------------------------
print()
print("=== ПРОВЕРКА ===")
print("sumif:                  ", a)
print("прямой цикл:            ", s)
print("обратный цикл + delete: ", k)
print("Все совпадают:", a == s and s == k)
