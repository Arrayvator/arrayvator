# ============================================================
# ДЕМО: countif  vs  цикл for  vs  обратный цикл + delete
# ============================================================
# Задача: посчитать количество сотрудников отдела "IT"
# ============================================================

# ------------------------------------------------------------
# 1. ИСХОДНЫЕ ДАННЫЕ
# ------------------------------------------------------------
m = ["Отдел", "Сотрудник", "Зарплата";
     "IT",    "Аня",       85000;
     "HR",    "Боб",       65000;
     "IT",    "Света",     95000;
     "Sales", "Гоша",      80000;
     "IT",    "Даша",      70000;
     "HR",    "Егор",      72000]

print("=== ИСХОДНАЯ ТАБЛИЦА ===")
print(m)


# ------------------------------------------------------------
# 2. СПОСОБ 1 — countif (декларативно, в одну строку)
# ------------------------------------------------------------
print()
print("=== СПОСОБ 1: countif ===")

a = countif(m[:, "Отдел"] == "IT")
print("IT-сотрудников:", a)


# ------------------------------------------------------------
# 3. СПОСОБ 2 — прямой цикл for (считаем вручную)
# ------------------------------------------------------------
print()
print("=== СПОСОБ 2: цикл for ===")

n = 0
for i = 2 to lenrow(m) do
{
    if m[i, "Отдел"] == "IT" then
    {
        n = n + 1
    }
}

print("IT-сотрудников:", n)


# ------------------------------------------------------------
# 4. СПОСОБ 3 — обратный цикл + delete (считаем через удаление)
# ------------------------------------------------------------
print()
print("=== СПОСОБ 3: обратный цикл + delete ===")

# 4.1. Копируем матрицу (присваивание делает копию)
b = m

# 4.2. Запоминаем исходное количество строк данных
total = lenrow(b) - 1    # -1 — заголовок

# 4.3. Идём С КОНЦА ВВЕРХ и удаляем всё, что НЕ "IT"
for i = lenrow(b) to 2 do
{
    if b[i, "Отдел"] != "IT" then
    {
        b = delete(b[i, :])
    }
}

# 4.4. Осталось только IT — считаем
k = lenrow(b) - 1    # -1 — заголовок
print("IT-сотрудников:", k)


# ------------------------------------------------------------
# 5. ПРОВЕРКА: ВСЕ ТРИ РЕЗУЛЬТАТА ДОЛЖНЫ СОВПАДАТЬ
# ------------------------------------------------------------
print()
print("=== ПРОВЕРКА ===")
print("countif:              ", a)
print("прямой цикл:          ", n)
print("обратный цикл + delete:", k)
print("Все совпадают:", a == n and n == k)
