# ============================================================
# ТЕСТ 1: медиана всей колонки (скаляр)
# ============================================================
m = ["Отдел", "Сотрудник", "Зарплата";
     "IT",    "Аня",       85000;
     "HR",    "Боб",       65000;
     "IT",    "Света",     95000;
     "Sales", "Гоша",      80000;
     "HR",    "Лена",      72000;
     "IT",    "Дима",      78000]

print("=== ИСХОДНАЯ МАТРИЦА ===")
print(m)
println("")

a = median(m[:, "Зарплата"])

print("Медиана всех зарплат:")
print(a)
println("")

# ============================================================
# ТЕСТ 2: медиана по группам (вектор)
# ============================================================
b = median(m[:, "Зарплата"], by m[:, "Отдел"])

print("Медиана по отделам (вектор, по строкам):")
print(b)
println("")

c = addcolumn(m, "Медиана_отдела", b)

print("=== МАТРИЦА С МЕДИАНОЙ ПО ОТДЕЛАМ ===")
print(c)
