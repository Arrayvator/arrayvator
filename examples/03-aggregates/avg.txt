# ТЕСТ 1: среднее всей колонки (скаляр)
# ============================================================
m = ["Отдел", "Сотрудник", "Зарплата";
     "IT",    "Аня",       85000;
     "HR",    "Боб",       65000;
     "IT",    "Света",     95000;
     "Sales", "Гоша",      80000;
     "HR",    "Лена",      72000;
     "IT",    "Дима",      78000]

print("=== ИСХОДНАЯ МАТРИЦА ===")
print(m)
println("")

a = avg(m[:, "Зарплата"])

print("Среднее всех зарплат:")
print(a)
println("")

# ============================================================
# ТЕСТ 2: среднее по группам (вектор)
# ============================================================
b = avg(m[:, "Зарплата"], by m[:, "Отдел"])

print("Среднее по отделам (вектор, по строкам):")
print(b)
println("")

c = addcolumn(m, "Среднее_отдела", b)

print("=== МАТРИЦА СО СРЕДНИМ ПО ОТДЕЛАМ ===")
print(c)
