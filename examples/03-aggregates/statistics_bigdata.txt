# ОБЩИЕ ДАННЫЕ
# ============================================================
m = ["Отдел", "Зарплата";
     "IT",    85000;
     "HR",    None;
     "IT",    95000;
     "Sales", 80000;
     "HR",    72000;
     "IT",    None]

bd = ToBigData(m)

# ============================================================
# 1. count — непустые
# ============================================================
print("=== count (непустые) ===")
print("Matrix:", count(m[:, "Зарплата"]))      # 4
print("DuckDB:", count(bd[:, "Зарплата"]))     # 4

# ============================================================
# 2. count by — непустые по группам
# ============================================================
println("")
print("=== count by (непустые по группам) ===")
print("Matrix:", count(m[:, "Зарплата"], by m[:, "Отдел"]))
print("DuckDB:", count(bd[:, "Зарплата"], by bd[:, "Отдел"]))
# ожидаем: [2, 1, 2, 1, 1, 2]

# ============================================================
# 3. count(by) — всего строк в группе
# ============================================================
println("")
print("=== count(by) (всего строк в группе) ===")
print("Matrix:", count(by m[:, "Отдел"]))
print("DuckDB:", count(by bd[:, "Отдел"]))
# ожидаем: [3, 2, 3, 1, 2, 3]

# ============================================================
# 4. sum — игнорирует None
# ============================================================
println("")
print("=== sum (игнорирует None) ===")
print("Matrix:", sum(m[:, "Зарплата"]))        # 252000
print("DuckDB:", sum(bd[:, "Зарплата"]))       # 252000

# ============================================================
# 5. sum by
# ============================================================
println("")
print("=== sum by ===")
print("Matrix:", sum(m[:, "Зарплата"], by m[:, "Отдел"]))
print("DuckDB:", sum(bd[:, "Зарплата"], by bd[:, "Отдел"]))

# ============================================================
# 6. avg
# ============================================================
println("")
print("=== avg ===")
print("Matrix:", avg(m[:, "Зарплата"]))        # 252000 / 4 = 63000.0
print("DuckDB:", avg(bd[:, "Зарплата"]))       # 63000.0

# ============================================================
# 7. min / max
# ============================================================
println("")
print("=== min / max ===")
print("Matrix min:", min(m[:, "Зарплата"]))    # 72000
print("DuckDB min:", min(bd[:, "Зарплата"]))   # 72000
print("Matrix max:", max(m[:, "Зарплата"]))    # 95000
print("DuckDB max:", max(bd[:, "Зарплата"]))   # 95000
