m = ["Отдел", "Сотрудник", "Зарплата";
     "IT",    "Аня",       85000;
     "HR",    "Боб",       65000;
     "IT",    "Света",     95000;
     "Sales", "Гоша",      80000;
     "HR",    "Лена",      72000;
     "IT",    "Дима",      78000]

print("=== ИСХОДНАЯ МАТРИЦА ===")
print(m)
println("")

a = min(m[:, "Зарплата"])

print("Минимум всех зарплат:")

println("")

# ============================================================
# ТЕСТ 2: минимум по группам (вектор)
# ============================================================
b = min(m[:, "Зарплата"], by m[:, "Отдел"])

print("Минимум по отделам (вектор, по строкам):")
print(b)
println("")

c = addcolumn(m, "Мин_отдела", b)

print("=== МАТРИЦА С МИНИМУМОМ ПО ОТДЕЛАМ ===")
print(c)
