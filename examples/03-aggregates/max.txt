# ТЕСТ 1: максимум всей колонки (скаляр)
# ============================================================
m = ["Отдел", "Сотрудник", "Зарплата";
     "IT",    "Аня",       85000;
     "HR",    "Боб",       65000;
     "IT",    "Света",     95000;
     "Sales", "Гоша",      80000;
     "HR",    "Лена",      72000;
     "IT",    "Дима",      78000]

print("=== ИСХОДНАЯ МАТРИЦА ===")
print(m)
println("")

a = max(m[:, "Зарплата"])

print("Максимум всех зарплат:")
print(a)
println("")

# ============================================================
# ТЕСТ 2: максимум по группам (вектор)
# ============================================================
b = max(m[:, "Зарплата"], by m[:, "Отдел"])

print("Максимум по отделам (вектор, по строкам):")
print(b)
println("")

c = addcolumn(m, "Макс_отдела", b)

print("=== МАТРИЦА С МАКСИМУМОМ ПО ОТДЕЛАМ ===")
print(c)
