# ТЕСТ 1: сумма всей колонки (скаляр)
# ============================================================
m = ["Отдел", "Сотрудник", "Зарплата";
     "IT",    "Аня",       85000;
     "HR",    "Боб",       65000;
     "IT",    "Света",     95000;
     "Sales", "Гоша",      80000;
     "HR",    "Лена",      72000;
     "IT",    "Дима",      78000]

print("=== ИСХОДНАЯ МАТРИЦА ===")
print(m)
println("")

a = sum(m[:, "Зарплата"])

print("Сумма всех зарплат:")
print(a)
println("")

# ============================================================
# ТЕСТ 2: сумма по группам (вектор)
# ============================================================
s = sum(m[:, "Зарплата"], by m[:, "Отдел"])

print("Сумма по отделам (вектор, по строкам):")
print(s)
println("")

c = addcolumn(m, "Сумма отдела", s)

print("=== МАТРИЦА С СУММОЙ ПО ОТДЕЛАМ ===")
print(c)
