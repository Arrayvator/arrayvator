# ============================================================
# ГРУППА 1 (усложнённая): УСЛОВНЫЕ АГРЕГАТЫ
# avgif / minif / maxif / medianif / countuniqueif / sumproduct
# с and / or / not
# ============================================================

m = ["Отдел", "Сотрудник", "Зарплата", "Возраст", "Город";
     "IT",    "Аня",       85000,      25,        "Москва";
     "IT",    "Боб",       95000,      32,        "Питер";
     "IT",    "Света",     70000,      28,        "Москва";
     "HR",    "Гена",      65000,      41,        "Москва";
     "HR",    "Даша",      72000,      35,        "Питер";
     "Sales", "Егор",      80000,      29,        "Москва";
     "Sales", "Жанна",     88000,      38,        "Питер";
     "Sales", "Зина",      60000,      27,        "Москва"]

print("=== ИСХОДНАЯ ===")
print(m)
print("")

# ------------------------------------------------------------
# 1. avgif с AND: IT и возраст > 26
# ------------------------------------------------------------
print("--- 1. avgif(m[:, 'Отдел'] == 'IT' and m[:, 'Возраст'] > 26, m[:, 'Зарплата']) ---")
r = avgif(m[:, "Отдел"] == "IT" and m[:, "Возраст"] > 26, m[:, "Зарплата"])
print(r)
# IT: Аня (25) — НЕ подходит, Боб (32) — да, Света (28) — да
# (95000 + 70000) / 2 = 82500
print("")

# ------------------------------------------------------------
# 2. minif с OR: IT или HR
# ------------------------------------------------------------
print("--- 2. minif(m[:, 'Отдел'] == 'IT' or m[:, 'Отдел'] == 'HR', m[:, 'Зарплата']) ---")
r = minif(m[:, "Отдел"] == "IT" or m[:, "Отдел"] == "HR", m[:, "Зарплата"])
print(r)
# IT: 85000, 95000, 70000; HR: 65000, 72000 → min = 65000
print("")

# ------------------------------------------------------------
# 3. maxif с AND: IT и Москва
# ------------------------------------------------------------
print("--- 3. maxif(m[:, 'Отдел'] == 'IT' and m[:, 'Город'] == 'Москва', m[:, 'Зарплата']) ---")
r = maxif(m[:, "Отдел"] == "IT" and m[:, "Город"] == "Москва", m[:, "Зарплата"])
print(r)
# IT + Москва: Аня (85000), Света (70000) → max = 85000
print("")

# ------------------------------------------------------------
# 4. medianif с OR: Sales или HR
# ------------------------------------------------------------
print("--- 4. medianif(m[:, 'Отдел'] == 'Sales' or m[:, 'Отдел'] == 'HR', m[:, 'Зарплата']) ---")
r = medianif(m[:, "Отдел"] == "Sales" or m[:, "Отдел"] == "HR", m[:, "Зарплата"])
print(r)
# Sales: 80000, 88000, 60000; HR: 65000, 72000
# отсортировано: [60000, 65000, 72000, 80000, 88000] → медиана = 72000
print("")

# ------------------------------------------------------------
# 5. countuniqueif с AND: IT и возраст > 26
# ------------------------------------------------------------
print("--- 5. countuniqueif(m[:, 'Отдел'] == 'IT' and m[:, 'Возраст'] > 26, m[:, 'Город']) ---")
r = countuniqueif(m[:, "Отдел"] == "IT" and m[:, "Возраст"] > 26, m[:, "Город"])
print(r)
# IT + возраст > 26: Боб (32, Питер), Света (28, Москва)
# уникальных городов = 2
print("")

# ------------------------------------------------------------
# 6. avgif с NOT: не IT
# ------------------------------------------------------------
print("--- 6. avgif(not (m[:, 'Отдел'] == 'IT'), m[:, 'Зарплата']) ---")
r = avgif(not (m[:, "Отдел"] == "IT"), m[:, "Зарплата"])
print(r)
# НЕ IT: HR (65000, 72000) + Sales (80000, 88000, 60000) = 365000 / 5 = 73000
print("")

# ------------------------------------------------------------
# 7. sumproduct — оставляем как есть (без условия)
# ------------------------------------------------------------
print("--- 7. sumproduct(m[:, 'Зарплата'], m[:, 'Возраст']) ---")
r = sumproduct(m[:, "Зарплата"], m[:, "Возраст"])
print(r)

print("=== ГРУППА 1 (усложнённая) ЗАВЕРШЕНА ===")
