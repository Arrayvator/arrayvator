# ============================================================
# ТЕСТ 1: std всей колонки (скаляр, округлён)
# ============================================================
m = ["Отдел", "Сотрудник", "Зарплата";
     "IT",    "Аня",       2;
     "HR",    "Боб",       4;
     "IT",    "Света",     4;
     "Sales", "Гоша",      6;
     "HR",    "Лена",      8;
     "IT",    "Дима",      4]

print("=== ИСХОДНАЯ МАТРИЦА ===")
print(m)
println("")

a = round(std(m[:, "Зарплата"]), 4)

print("Стандартное отклонение всех значений:")
print(a)
println("")

# ============================================================
# ТЕСТ 2: std по группам (вектор, округлён)
# ============================================================
b = round(std(m[:, "Зарплата"], by m[:, "Отдел"]), 2)

print("Стандартное отклонение по отделам (вектор, по строкам):")
print(b)
println("")

c = addcolumn(m, "Отклонение", b)

print("=== МАТРИЦА С ОТКЛОНЕНИЕМ ПО ОТДЕЛАМ ===")
print(c)
