# Агрегаты по группам

# Исходная матрица
m = ["Отдел", "Сотрудник", "Зарплата";
     "IT",    "Аня",       85000;
     "HR",    "Боб",       65000;
     "IT",    "Света",     95000;
     "Sales", "Гоша",      80000;
     "HR",    "Лена",      72000;
     "IT",    "Дима",      78000]
println(m)
# После:
# Отдел  Сотрудник  Зарплата
# IT     Аня        85000
# HR     Боб        65000
# IT     Света      95000
# Sales  Гоша       80000
# HR     Лена       72000
# IT     Дима       78000


# sum by Отдел
m = addcolumn(m, "Сумма", sum(m[:, "Зарплата"], by m[:, "Отдел"]))


# avg by Отдел
m = addcolumn(m, "Среднее", round(avg(m[:, "Зарплата"], by m[:, "Отдел"]), 2))


# min by Отдел
m = addcolumn(m, "Минимум", min(m[:, "Зарплата"], by m[:, "Отдел"]))


# max by Отдел
m = addcolumn(m, "Максимум", max(m[:, "Зарплата"], by m[:, "Отдел"]))


# median by Отдел
m = addcolumn(m, "Медиана", round(median(m[:, "Зарплата"], by m[:, "Отдел"]), 2))


# std by Отдел
m = addcolumn(m, "Отклонение", round(std(m[:, "Зарплата"], by m[:, "Отдел"]), 2))


# count by Отдел
m = addcolumn(m, "Количество", count(m[:, "Зарплата"], by m[:, "Отдел"]))


# Итоговая таблица
println(m)
# После:
# Отдел  Сотрудник  Зарплата  Сумма   Среднее  Минимум  Максимум  Медиана  Отклонение  Количество

# HR     Боб        65000     137000  68500.0  65000    72000     68500.0  3500.0      2
# IT     Света      95000     258000  86000.0  78000    95000     85000    6976.15     3
# Sales  Гоша       80000     80000   80000.0  80000    80000     80000    0           1
# HR     Лена       72000     137000  68500.0  65000    72000     68500.0  3500.0      2
# IT     Дима       78000     258000  86000.0  78000    95000     85000    6976.15     3
