# ============================================================
# ТЕСТ 1: количество непустых (скаляр)
# ============================================================
m = ["Отдел", "Сотрудник", "Зарплата";
     "IT",    "Аня",       85000;
     "HR",    "Боб",       None;
     "IT",    "Света",     95000;
     "Sales", "Гоша",      80000;
     "HR",    "Лена",      72000;
     "IT",    "Дима",      None]

print("=== ИСХОДНАЯ МАТРИЦА ===")
print(m)
println("")

a = count(m[:, "Зарплата"])

print("Непустых значений в столбце Зарплата:")
print(a)
println("")

# ============================================================
# ТЕСТ 2: количество непустых в группе (вектор)
# ============================================================
b = count(m[:, "Зарплата"], by m[:, "Отдел"])

print("Непустых по отделам (вектор, по строкам):")
print(b)
println("")

c = addcolumn(m, "Заполнено_в_отделе", b)

print("=== МАТРИЦА: НЕПУСТЫЕ ПО ОТДЕЛАМ ===")
print(c)
println("")

# ============================================================
# ТЕСТ 3: количество строк в группе (вектор, без аргумента)
# ============================================================
d = count(by m[:, "Отдел"])

print("Всего строк по отделам (вектор, по строкам):")
print(d)
println("")

e = addcolumn(c, "Всего_в_отделе", d)

print("=== МАТРИЦА: ВСЕ СТРОКИ ПО ОТДЕЛАМ ===")
print(e)
