# ============================================================
# ДЕМО: расчёт стажа и надбавки за выслугу лет, один год = одному проценту надбавки
# ============================================================

# 1. Исходная матрица
m = [
    "Сотрудник", "Приём",      "Увольнение", "Зарплата";
    "Аня",       "15.03.2021", "",           80000;
    "Боб",       "01.07.2019", "30.06.2024", 80000;
    "Света",     "10.01.2022", "",           80000;
    "Гена",      "20.09.2018", "",           80000;
    "Даша",      "05.12.2015", "",           80000;
    "Игорь",     "01.02.2023", "",           80000;
    "Катя",      "10.06.2020", "31.12.2023", 80000
]

print("=== ИСХОДНАЯ ТАБЛИЦА ===")
print(m)
println()

# Добавляем столбец "Конечная_дата и Стаж, т.к. таких столбцов нет, они автоматом создаются"
m[1,"Конечная дата"]=0
m[1,"Стаж"]=0

# Заменяем в колонке Конечная дата, текущую дату на сегодня, кроме тех, которые уволились
m=applyif(m[:,"Увольнение"]=="", m[:,"Конечная дата"]=datenow("DD.MM.YYYY"))

# Считаем стаж в полных годах, функция DataDiff
m[:,"Стаж"]=DateDiff(m[:, "Приём"], m[:,"Конечная дата"], "DD.MM.YYYY", "years")

# Считаем надбавку для сотрудника
m[:,"ЗП с надбавкой"]=m[:, "Зарплата"]+((m[:,"Стаж"]/100)*m[:, "Зарплата"])
println()
print(m) 

# Удаляем строки, где сотрудники уволились (end это последний столбец или "ЗП с надбавкой")
m=deleteif(m[:, end]==None)
m[:,"Полный стаж"]=DateDiff(m[:, "Приём"], m[:,"Конечная дата"], "DD.MM.YYYY", "all")

println()
print(m)
