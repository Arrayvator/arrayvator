# 2. Начинаем отчёт

report("Продажи магазина электроники", "electronics_report.html")

# ============================================================
# 3. Сводка по городам
# ============================================================
report_section("1. Сводка по городам")

cities = Unique(m[:, "Город"])
s = addcolumn(cities, "Сумма", sum(m[:, "Сумма"], by m[:, "Город"]))
s = addcolumn(s, "Доля_%", percentof(s[:, "Сумма"]))
s = sort(s[:, "Сумма"], ZA)

report_table(s, title "Продажи по городам")

report_chart(bar,
             s[:, "Город"],
             s[:, "Сумма"],
             title "Продажи по городам",
             xlabel "Город",
             ylabel "Сумма, руб.")

# ============================================================
# 4. Динамика по месяцам
# ============================================================
report_section("2. Динамика продаж")

months = Unique(m[:, "Месяц"])
ms = addcolumn(months, "Сумма", sum(m[:, "Сумма"], by m[:, "Месяц"]))

report_chart(line,
             ms[:, "Месяц"],
             ms[:, "Сумма"],
             title "Продажи по месяцам",
             xlabel "Месяц",
             ylabel "Сумма, руб.")

# ============================================================
# 5. Доли городов
# ============================================================

             s[:, "Город"],

# ============================================================
# 6. Каналы продаж
# ============================================================
report_section("4. Каналы продаж")

             cs[:, "Канал"],

             title "Продажи по каналам")

# ============================================================
