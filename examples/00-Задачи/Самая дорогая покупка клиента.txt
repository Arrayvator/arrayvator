# ДЕМО: самая дорогая покупка клиента
#        (сумма + название товара) — всё в одной таблице
# ============================================================

m = [
    "Клиент", "Товар",     "Сумма";
    "Аня",    "Яблоко",    100;
    "Боб",    "Курица",    200;
    "Аня",    "Банан",     120;
    "Света",  "Говядина",  300;
    "Боб",    "Груша",     90;
    "Света",  "Свёкла",    40;
    "Аня",    "Свинина",   250;
    "Боб",    "Огурец",    60;
    "Света",  "Апельсин",  110
]

print("=== ИСХОДНАЯ ТАБЛИЦА ===")
print(m)
print("")
# 1. Максимальная сумма каждого клиента
m = firstvalue(m[:, "Сумма"], by m[:, "Клиент"],
               order m[:, "Сумма"], ZA)

# 2. Название товара с максимальной суммой
m = firstvalue(m[:, "Товар"], by m[:, "Клиент"],
               order m[:, "Сумма"], ZA)

# 3. Понятные имена столбцов
m[1, end-1] = "Макс сумма"
m[1, end]   = "Макс товар"

print("=== РЕЗУЛЬТАТ ===")
print(m)
println()
