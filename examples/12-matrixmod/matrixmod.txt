# ============================================================
# ТЕСТЫ MATRIXMOD — рабочие сценарии
# ============================================================

m = ["Имя", "Отдел", "Зарплата";
     "Аня", "IT",    85000;
     "Боб", "HR",    65000;
     "Света","IT",   95000;
     "Гена","Sales", 70000;
     "Даша","IT",    88000]


# ------------------------------------------------------------
# 1. DELETE — одну строку
# ------------------------------------------------------------
println("=== 1. matrixmod(m, delete, 2) ===")
r = matrixmod(m, delete, 2)
println(r)
println("")


# ------------------------------------------------------------
# 2. DELETE — несколько строк
# ------------------------------------------------------------
println("=== 2. matrixmod(m, delete, [2, 4]) ===")
r = matrixmod(m, delete, [2, 4])
println(r)
println("")


# ------------------------------------------------------------
# 3. DELETE — last 2
# ------------------------------------------------------------
println("=== 3. matrixmod(m, delete, last 2) ===")
r = matrixmod(m, delete, "last 2")
println(r)
println("")


# ------------------------------------------------------------
# 4. INSERT before
# ------------------------------------------------------------
println("=== 4. matrixmod(m, insert, 3, before) ===")
r = matrixmod(m, insert, 3, before)
println(r)
println("")


# ------------------------------------------------------------
# 5. INSERT after
# ------------------------------------------------------------
println("=== 5. matrixmod(m, insert, 2, after) ===")
r = matrixmod(m, insert, 2, after)
println(r)
println("")


# ------------------------------------------------------------
# 6. INSERT — несколько строк
# ------------------------------------------------------------
println("=== 6. matrixmod(m, insert, [2, 4], after) ===")
r = matrixmod(m, insert, [2, 4], after)
println(r)
println("")


# ------------------------------------------------------------
# 7. DUPLICATE before
# ------------------------------------------------------------
println("=== 7. matrixmod(m, duplicate, 2, before) ===")
r = matrixmod(m, duplicate, 2, before)
println(r)
println("")


# ------------------------------------------------------------
# 8. DUPLICATE after
# ------------------------------------------------------------
println("=== 8. matrixmod(m, duplicate, 3, after) ===")
r = matrixmod(m, duplicate, 3, after)
println(r)
println("")


# ------------------------------------------------------------
# 9. CLEAR одну строку
# ------------------------------------------------------------
println("=== 9. matrixmod(m, clear, 2) ===")
r = matrixmod(m, clear, 2)
println(r)
println("")


# ------------------------------------------------------------
# 10. CLEAR несколько
# ------------------------------------------------------------
println("=== 10. matrixmod(m, clear, [2, 3]) ===")
r = matrixmod(m, clear, [2, 3])
println(r)
println("")


# ------------------------------------------------------------
# 11. KEEP [1, 3, 5]
# ------------------------------------------------------------
println("=== 11. matrixmod(m, keep, [1, 3, 5]) ===")
r = matrixmod(m, keep, [1, 3, 5])
println(r)
println("")


# ------------------------------------------------------------
# 12. KEEP [1, 2, 4]
# ------------------------------------------------------------
println("=== 12. matrixmod(m, keep, [1, 2, 4]) ===")
r = matrixmod(m, keep, [1, 2, 4])
println(r)
println("")


# ------------------------------------------------------------
# 13. SWAP [2, 5]
# ------------------------------------------------------------
println("=== 13. matrixmod(m, swap, [2, 5]) ===")
r = matrixmod(m, swap, [2, 5])
println(r)
println("")


# ------------------------------------------------------------
# 16. DELETE end
# ------------------------------------------------------------
println("=== 16. matrixmod(m, delete, end) ===")
r = matrixmod(m, delete, end)
println(r)
println("")


# ------------------------------------------------------------
# 17. DELETE end-1
# ------------------------------------------------------------
println("=== 17. matrixmod(m, delete, end-1) ===")
r = matrixmod(m, delete, end-1)
println(r)
println("")


# ------------------------------------------------------------
# 18. Мутация
# ------------------------------------------------------------
println("=== 18. мутация m2 = matrixmod(...) ===")
m2 = ["A", "B"; 1, 2; 3, 4]
m2 = matrixmod(m2, delete, 2)
println(m2)
println("")


# ------------------------------------------------------------
# 19. DuckDB — ожидается ошибка выполнения (можно поймать)
# ------------------------------------------------------------
println("=== 19. matrixmod на DuckDB — ожидается ошибка ===")
m3 = ["A", "B"; 1, 2; 3, 4]
bd = ToBigData(m3)

try {
    r = matrixmod(bd, delete, 2)
    println("❌ НЕ ДОЛЖНО БЫТЬ: matrixmod не упал")
} catch {
    println("✅ OK: matrixmod упал с понятной ошибкой")
}
println("")


println("=== РАБОЧИЕ ТЕСТЫ MATRIXMOD ЗАВЕРШЕНЫ ===")
