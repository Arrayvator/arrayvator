# ============================================================
# REPORT — тест: матрица + все графики → HTML
# ============================================================

m = ["Месяц", "Продажи", "Затраты";
     "Янв",   100,       60;
     "Фев",   120,       70;
     "Мар",   90,        55;
     "Апр",   140,       80;
     "Май",   160,       90;
     "Июн",   180,       95]

print("=== ИСХОДНАЯ МАТРИЦА ===")
print(m)
print("")

# ============================================================
# 1. Начать отчёт
# ============================================================
report("Анализ продаж", "report_all.html")
print("report: started")
print("")

# ============================================================
# 2. Секция 1 — исходные данные
# ============================================================
report_section("1. Исходные данные")
print("section 1: OK")
print("")

report_text("Ниже приведены данные по продажам за 6 месяцев.")
print("text: OK")
print("")

report_table(m, title "Таблица продаж")
print("table: OK")
print("")

# ============================================================
# 3. Секция 2 — графики
# ============================================================
report_section("2. Графики")
print("section 2: OK")
print("")

report_chart(bar, m[:, "Месяц"], m[:, "Продажи"],
             title "Продажи по месяцам")
print("chart bar: OK")
print("")

report_chart(line, m[:, "Месяц"], m[:, "Продажи"],
             title "Динамика продаж")
print("chart line: OK")
print("")

report_chart(pie, m[:, "Месяц"], m[:, "Продажи"],
             title "Доли продаж")
print("chart pie: OK")
print("")

report_chart(hist, m[:, "Продажи"], bins 5,
             title "Гистограмма продаж")
print("chart hist: OK")
print("")

report_chart(scatter, m[:, "Продажи"], m[:, "Затраты"],
             title "Продажи vs Затраты")
print("chart scatter: OK")
print("")

report_chart(box, m[:, "Месяц"], m[:, "Продажи"],
             title "Распределение продаж")
print("chart box: OK")
print("")

# ============================================================
# 4. Секция 3 — сводка
# ============================================================
report_section("3. Сводка")
print("section 3: OK")
print("")

report_text("Всего продаж: " + to_string(sum(m[:, "Продажи"])))
print("summary: OK")
print("")

# ============================================================
# 5. Сохранить
# ============================================================
report_save(true)
print("save: OK")
print("")

print("=== ОТЧЁТ СОХРАНЁН ===")
