# ============================================================
# УЧЕБНЫЙ ПРИМЕР: 4 способа на ХАОТИЧНЫХ данных
#
# ЗАДАЧА:
#   Посчитать сумму зарплат по отделам.
#
#   Данные в ХАОТИЧНОМ порядке (отделы вперемешку).
#   Сравниваем 4 способа:
#     1. groupby                   — сводка
#     2. addcolumn + sumif(by)     — сумма в каждой строке
#     3. pivot                     — сводка (другой вид)
#     4. groupagg                  — детали + итог (требует сортировки!)
# ============================================================

# ------------------------------------------------------------
# ХАОТИЧНЫЕ ДАННЫЕ — отделы вперемешку
# ------------------------------------------------------------
m = ["Отдел", "Сотрудник", "Зарплата";
     "IT",    "Аня",       85000;
     "HR",    "Егор",      65000;
     "IT",    "Боб",       95000;
     "Sales", "Катя",      80000;
     "HR",    "Жанна",     72000;
     "IT",    "Света",     70000;
     "Sales", "Лена",      88000;
     "HR",    "Зина",      58000;
     "IT",    "Гена",      88000;
     "Sales", "Миша",      60000;
     "IT",    "Даша",      62000;
     "HR",    "Игорь",     81000;
     "Sales", "Нина",      95000;
     "Sales", "Олег",      55000;
     "Sales", "Петя",      73000]

print("=" * 60)
print("ИСХОДНАЯ (хаос — отделы вперемешку)")
print("=" * 60)
print(m)
print("")


# ============================================================
# СПОСОБ 1: groupby — сводка
# ============================================================
print("=" * 60)
print("СПОСОБ 1: groupby (сводка)")
print("=" * 60)
print("groupby группирует ПО ЗНАЧЕНИЮ.")
print("Сортировка НЕ нужна.")
print("")

r1 = groupby(by m[:, "Отдел"], agg sum(m[:, "Зарплата"]))
print(r1)
print("")


# ============================================================
# СПОСОБ 2: addcolumn + sumif(by) — сумма в каждой строке
# ============================================================
print("=" * 60)
print("СПОСОБ 2: addcolumn + sumif(by) (сумма в каждой строке)")
print("=" * 60)
print("sumif(by ...) группирует ПО ЗНАЧЕНИЮ.")
print("Сортировка НЕ нужна.")
print("")

r2 = addcolumn(m, "Сумма_отд", sumif(by m[:, "Отдел"], m[:, "Зарплата"]))
print(r2)
print("")


# ============================================================
# СПОСОБ 3: pivot — сводка (другой вид)
# ============================================================
print("=" * 60)
print("СПОСОБ 3: pivot (сводка)")
print("=" * 60)
print("pivot группирует ПО ЗНАЧЕНИЮ.")
print("Сортировка НЕ нужна.")
print("")

r3 = pivot(m[:, "Отдел"], m[:, "Зарплата"], sum)
print(r3)
print("")


# ============================================================
# СПОСОБ 4: groupagg — детали + итог
# ============================================================
print("=" * 60)
print("СПОСОБ 4: groupagg (детали + итог)")
print("=" * 60)
print("groupagg режет ПО ПОРЯДКУ строк.")
print("С хаотичными данными — много блоков, неправильные итоги.")
print("")

r4 = groupagg(m[:, "Отдел"], m[:, "Зарплата"], sum, "ИТОГО")
print(r4)
print("")


# ============================================================
# СПОСОБ 4-БИС: сортировка AZ + groupagg
# ============================================================
print("=" * 60)
print("СПОСОБ 4-БИС: sort(AZ) + groupagg")
print("=" * 60)
print("Сначала сортируем — одинаковые значения идут подряд.")
print("Потом groupagg даёт правильные блоки.")
print("")

# ШАГ 1: сортируем по столбцу "Отдел"
m_sorted = sort(m[:, "Отдел"], AZ)

print("--- отсортированная таблица ---")
print(m_sorted)
print("")

# ШАГ 2: применяем groupagg
r4b = groupagg(m_sorted[:, "Отдел"], m_sorted[:, "Зарплата"], sum, "ИТОГО")
print("--- groupagg на отсортированных данных ---")
print(r4b)
print("")


# ============================================================
# СРАВНЕНИЕ
# ============================================================
print("=" * 60)
print("СРАВНЕНИЕ")
print("=" * 60)
print("")
print("Правильные итоги (по 3 отделам):")
print("  HR:     276000")
print("  IT:     400000")
print("  Sales:  451000")
print("")
print("Общая сумма всех зарплат: 1127000")
print("")


# ============================================================
# ВЫВОД
# ============================================================
print("=" * 60)
print("ВЫВОД")
print("=" * 60)
print("")
print("groupby / addcolumn+sumif / pivot:")
print("     → группируют ПО ЗНАЧЕНИЮ, сортировка НЕ нужна.")
print("")
print("groupagg:")
print("     → режет ПО ПОРЯДКУ, требует сортировки AZ.")
print("     → без сортировки даёт 'рваные' блоки.")
print("")
print("Правильный рецепт для groupagg:")
print("     m_sorted = sort(m[:, \"Отдел\"], AZ)")
print("     r = groupagg(m_sorted[:, \"Отдел\"], m_sorted[:, \"Зарплата\"], sum)")
print("")
