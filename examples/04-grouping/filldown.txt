m = ["Клиент", "Товар";
     "Аня",    "Яблоки";
     "",       "Груши";
     "",       "Сливы";
     "Боб",    "Бананы";
     "",       "Апельсины"]

print("=== ИСХОДНАЯ ===")
print(m)
print("")

print("=== filldown(m[:, 'Клиент']) ===")
r = filldown(m[:, "Клиент"])
print(r)
print("")

print("=== filldown с маркером '-' ===")
m2 = ["Клиент";
      "Аня";
      "-";
      "-";
      "Боб";
      "-"]

print(m2)
print()
r = filldown(m2[:, "Клиент"], "-")
print(r)
