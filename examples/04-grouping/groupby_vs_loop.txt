# ============================================================
# ДЕМО: groupby  vs  цикл for
# ============================================================
# Задача: сводная "Отдел → Сумма зарплат"
# ============================================================

# ------------------------------------------------------------
# 1. ИСХОДНЫЕ ДАННЫЕ
# ------------------------------------------------------------
m = ["Отдел", "Сотрудник", "Зарплата";
     "IT",    "Аня",       85000;
     "HR",    "Боб",       65000;
     "IT",    "Света",     95000;
     "Sales", "Гоша",      80000;
     "IT",    "Даша",      70000;
     "HR",    "Егор",      72000;
     "IT",    "Женя",      90000;
     "Sales", "Зина",      60000;
     "HR",    "Игорь",     80000;
     "Sales", "Костя",     70000;
     "IT",    "Лена",      65000;
     "HR",    "Миша",      90000]

print("=== ИСХОДНАЯ ТАБЛИЦА ===")
print(m)


# ------------------------------------------------------------
# 2. СПОСОБ 1 — groupby (одна строка)
# ------------------------------------------------------------
print()
print("=== СПОСОБ 1: groupby ===")

a = groupby(by m[:, "Отдел"], agg sum(m[:, "Зарплата"]))
print(a)


# ------------------------------------------------------------
# 3. СПОСОБ 2 — цикл for (вручную)
# ------------------------------------------------------------
print()
print("=== СПОСОБ 2: цикл for ===")

# 3.1. Собираем УНИКАЛЬНЫЕ отделы (порядок первого появления)
c = []
for i = 2 to lenrow(m) do
{
    d = m[i, "Отдел"]

    f = 0
    if len(c) > 0 then
    {
        for j = 1 to len(c) do
        {
            if c[j] == d then
            {
                f = 1
            }
        }
    }

    if f == 0 then
    {
        c[end+1] = d
    }
}

# 3.2. Создаём матрицу-результат
n = len(c)
b = matrix(n + 1, 2)

b[1, 1] = "Отдел"
b[1, 2] = "Сумма_Зарплата"

# 3.3. Для каждого отдела считаем сумму
k = 2
for i = 1 to n do
{
    d = c[i]

    s = 0
    for j = 2 to lenrow(m) do
    {
        if m[j, "Отдел"] == d then
        {
            s = s + m[j, "Зарплата"]
        }
    }

    b[k, 1] = d
    b[k, 2] = s
    k = k + 1
}

print(b)


# ------------------------------------------------------------
# 4. ПРОВЕРКА
# ------------------------------------------------------------
print()
print("=== ПРОВЕРКА ===")
print("Строк в a:", lenrow(a))
print("Строк в b:", lenrow(b))
print("Совпадают:", lenrow(a) == lenrow(b))
