# ============================================================
# ТЕСТ: groupagg на ХАОТИЧНЫХ данных
# ============================================================

# ------------------------------------------------------------
# ХАОТИЧНЫЕ ДАННЫЕ — отделы вперемешку
# ------------------------------------------------------------
m = ["Отдел", "Сотрудник", "Зарплата";
     "IT",    "Аня",       85000;
     "HR",    "Егор",      65000;
     "IT",    "Боб",       95000;
     "Sales", "Катя",      80000;
     "HR",    "Жанна",     72000;
     "IT",    "Света",     70000;
     "Sales", "Лена",      88000;
     "HR",    "Зина",      58000;
     "IT",    "Гена",      88000;
     "Sales", "Миша",      60000;
     "IT",    "Даша",      62000;
     "HR",    "Игорь",     81000;
     "Sales", "Нина",      95000;
     "Sales", "Олег",      55000;
     "Sales", "Петя",      73000]

print("=== ИСХОДНАЯ (хаос) ===")
print(m)
print("")


# ------------------------------------------------------------
# 1. groupagg БЕЗ сортировки — что получится?
# ------------------------------------------------------------
print("=== 1. groupagg БЕЗ сортировки ===")
print("Ожидание: подытог будет после КАЖДОГО перехода значения.")
print("")

r1 = groupagg(m[:, "Отдел"], m[:, "Зарплата"], sum, "ИТОГО")
print(r1)
print("")


# ------------------------------------------------------------
# 2. Сортируем и потом groupagg
# ------------------------------------------------------------
print("=== 2. Сортировка AZ + groupagg ===")

# Сортируем таблицу по столбцу "Отдел"
m_sorted = sort(m[:, "Отдел"], AZ)

print("--- отсортированная ---")
print(m_sorted)
print("")

r2 = groupagg(m_sorted[:, "Отдел"], m_sorted[:, "Зарплата"], sum, "ИТОГО")
print("--- groupagg на отсортированных данных ---")
print(r2)
print("")

print("=== ЗАВЕРШЁН ===")
