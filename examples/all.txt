     "Яблоки", 100;
     "Бананы", 80;

# for + ветвление: чётное/нечётное

     1,       "";
     2,       "";
     3,       "";

     6,       ""]

#   Присваивает строке ПЛОТНЫЙ ранг внутри группы.

#

# ФУНКЦИЯ: qualify
# ============================================================
# ЧТО ДЕЛАЕТ:
#   Фильтрует строки по результату оконной функции.
#   Работает КАК WHERE, но для оконных функций.
#   Оконная функция вычисляется ДО фильтрации.

#   Классика: "топ-3 по зарплате в каждом отделе".
#   Без qualify пришлось бы делать два шага:
#     1. rownumber(...) → новая таблица с столбцом
#     2. filterif(новая_таблица[:, "__rownumber"] <= 3)
#   С qualify — в одну строку.
#

# ============================================================
# ФУНКЦИЯ: winsum
# ============================================================
# ЧТО ДЕЛАЕТ:
#   Накопительная сумма. Каждая строка получает сумму
#   ВСЕХ предыдущих + текущей.
#
# ЗАЧЕМ НУЖНА:
#   Кумулятивные итоги: "общая выручка нарастающим итогом",
#   "сколько всего продано с начала года".
#   Для графиков накопления, для расчёта ABC.
#
# ОСОБЕННОСТИ:
#   - Порядок накопления задаётся через order.
#   - Без order — берётся порядок строк как есть.
#   - С by — накопление сбрасывается внутри каждой группы.
#
# СИНТАКСИС:
#   winsum(m[:, "X"] [, by ...] [, order ...] [, AZ | ZA])
#
# СТОЛБЕЦ РЕЗУЛЬТАТА: __winsum
# ============================================================

m = ["Отдел", "Сотрудник", "Зарплата", "Возраст", "Год";
     "IT",      "Аня",       85000,      25,       2023;
     "IT",      "Боб",       95000,      32,       2023;
     "IT",      "Света",     70000,      28,       2024;
     "HR",      "Гена",      65000,      41,       2023;
     "HR",      "Даша",      72000,      35,       2024;
     "Sales",   "Егор",      80000,      29,       2023;
     "Sales",   "Жанна",     88000,      38,       2024;
     "Sales",   "Зина",      60000,      22,       2024]


# ------------------------------------------------------------
# 1. Базовая накопительная сумма (порядок строк как есть)
# ------------------------------------------------------------
print("--- 1. winsum(Зарплата) ---")
r = winsum(m[:, "Зарплата"])
print(r)
print("")
# ОЖИДАНИЕ:
# 85000, 180000, 250000, 315000, 387000, 467000, 555000, 615000


# ------------------------------------------------------------
# 2. С сортировкой по ЗП (AZ) — нарастающий итог
# ------------------------------------------------------------
print("--- 2. winsum(Зарплата, order Зарплата, AZ) ---")
r = winsum(m[:, "Зарплата"], order m[:, "Зарплата"], AZ)
print(r)
print("")
# ОЖИДАНИЕ: накопление от 60000 к 95000,
# но в исходном порядке строк:
# Зина=60000, Гена=125000, Света=195000, Даша=267000,
# Егор=347000, Аня=432000, Жанна=520000, Боб=615000


# ------------------------------------------------------------
# 3. По группам (by Отдел) — накопление сбрасывается
# ------------------------------------------------------------
print("--- 3. winsum(Зарплата, by Отдел) ---")
r = winsum(m[:, "Зарплата"], by m[:, "Отдел"])
print(r)
print("")
# ОЖИДАНИЕ (в исходном порядке строк):
# IT:    Аня=85000, Боб=180000, Света=250000
# HR:    Гена=65000, Даша=137000
# Sales: Егор=80000, Жанна=168000, Зина=228000


# ------------------------------------------------------------
# 4. По группам + сортировка
# ------------------------------------------------------------
print("--- 4. winsum(Зарплата, by Отдел, order Зарплата, AZ) ---")
r = winsum(m[:, "Зарплата"],
           by m[:, "Отдел"],
           order m[:, "Зарплата"],
           AZ)
print(r)
print("")
# ОЖИДАНИЕ: внутри каждой группы — накопление по возрастанию ЗП:

n = 10000
big = matrix(n + 1, 3)
big[1, :] = ["ID", "Группа", "Сумма"]

for i = 2 to n + 1 do {
    big[i, 1] = i - 1
    big[i, 2] = (i % 5) + 1
    big[i, 3] = 1000 + (i % 500)
}

# Ещё одна матрица
big2 = matrix(1001, 2)
big2[1, :] = ["Группа", "Название"]

for i = 2 to 1001 do {
    big2[i, 1] = i
    big2[i, 2] = "Группа_" + to_string(i)
}

print("Матрица big2: " + to_string(lenrow(big2)) + " строк")

bd_big  = ToBigData(big)

# ============================================================
# Автосоздание столбцов и запрет автосоздания строк
# ============================================================
# Демонстрация двух важных правил ArrayVator:
#
#   1. Столбцы создаются АВТОМАТИЧЕСКИ по имени:
#          m[:, "Новый столбец"] = значение
#
#   2. Строки по имени НЕ создаются:
#          m["Новая строка", :] = значение    → ошибка
#
# Почему так:
#   • Столбцов обычно десятки — автосоздание безопасно и удобно.
#   • Строк могут быть миллионы — автосоздание опасно:
#     опечатка добавила бы мусорную строку в огромную таблицу.
# ============================================================


# ============================================================
# ПРИМЕР 1. Автосоздание столбца
# ============================================================

m = ["Товар",    "Продажи";
     "Яблоки",   120;
     "Бананы",   80;
     "Груши",    150]

# Столбца "Цена" ещё нет — он создастся справа
# и заполнится значением 100 во всех строках данных.
m[:, "Цена"] = 100

print("После добавления столбца 'Цена':")
print(m)
print()

# Ожидаемый вывод:
# Товар   Продажи  Цена
# Яблоки  120      100
# Бананы  80       100
# Груши   150      100


# ============================================================
# ПРИМЕР 2. Создать столбец с разными значениями (вектор)
# ============================================================

m[:, "Скидка"] = [5, 10, 15]

print("После добавления столбца 'Скидка' с вектором:")
print(m)
print()

# Ожидаемый вывод:
# Товар   Продажи  Цена  Скидка
# Яблоки  120      100   5
# Бананы  80       100   10
# Груши   150      100   15


# ============================================================
# ПРИМЕР 3. Записать в существующий столбец
# ============================================================

# Столбец "Цена" уже есть — значение перезапишется
# во всех строках данных.
m[:, "Цена"] = 200

print("После перезаписи столбца 'Цена':")
print(m)
print()


# ============================================================
# ПРИМЕР 4. Запрет автосоздания строки (должна быть ошибка)
# ============================================================

# Строки "Фрукт" нет — присваивание по имени НЕ создаёт строку.
# Будет ошибка ROW_NAME_NOT_FOUND.
#
# Раскомментируйте, чтобы увидеть ошибку:
#
# m["Фрукт", :] = 0
#
# ❌ Ошибка:
#   Строка с именем 'Фрукт' не найдена в первом столбце.
#   💡 Доступные имена: "Яблоки", "Бананы", "Груши"
#   💡 Если хотите ДОБАВИТЬ строку — используйте номер:
#        m[5, :] = ...


# ============================================================
# ПРИМЕР 5. Как ПРАВИЛЬНО добавить строку
# ============================================================

# Через end+1 — расширение вниз
m[end+1, :] = ["Фрукт", 300, 200, 20]

print("После добавления строки 'Фрукт' через end+1:")
print(m)
print()

# Ожидаемый вывод:
# Товар   Продажи  Цена  Скидка
# Яблоки  120      200   5
# Бананы  80       200   10
# Груши   150      200   15
# Фрукт   300      200   20


# ============================================================
# ПРИМЕР 6. Добавить строку через явный номер
# ============================================================

# Номер 10 — за пределами текущей матрицы (5 строк).
# Матрица расширится, промежуточные строки заполнятся None.
# m[10, :] = ["Киви", 50, 200, 0]


# ============================================================
# ПРИМЕР 7. Массовое добавление строк через addrows
# ============================================================

m = addrows(m, 2, 0)              # 2 пустые строки, все значения 0
m[end-1, 1] = "Манго"             # заполнить имя первой новой
m[end-1, 2] = 999                 # продажи
m[end, 1] = "Киви"                # имя второй новой
m[end, 2] = 555                   # продажи

print("После addrows (2 строки):")
print(m)
print()


# ============================================================
# ИТОГО
# ============================================================
# Столбцы:
#   m[:, "Новый"] = значение        — создать и заполнить
#   m[1, "Новый"] = значение        — то же самое
#   m[:, "Новый"] = [1, 2, 3]       — создать и заполнить вектором
#
# Строки:
#   m[end+1, :] = ["Имя", 100]      — добавить в конец
#   m[10, :] = ["Имя", 100]         — добавить по явному номеру
#   m = addrows(m, 1)               — добавить пустую
#   m = addrows(m, 1, 0)            — добавить с заполнением
#
# НЕ РАБОТАЕТ:
#   m["Новая строка", :] = 0        — ошибка ROW_NAME_NOT_FOUND
# ============================================================
# copy — только строки
# ============================================================

m = ["A", "B", "C";
     "Один", 2, 3;
     "Два", 5, 6;
     "Три", 7, 8, ]

print("Исходная:")
print(m)
print("")

print("copy(m[1, :], m[3, :], before):")
r = copy(m[1, :], m[3, :], before)
print(r)
print("")

print("copy(m[2, :], m[4, :], after):")
r = copy(m[2, :], m[4, :], after)
print(r)
print("")

print("copy(m[1:2, :], m[4, :], before):")
r = copy(m[1:2, :], m[4, :], before)
print(r)
print("")

print("copy(m['Один', :], m['Три', :], after):")
r = copy(m["Один", :], m["Три", :], after)
print(r)
print("")
